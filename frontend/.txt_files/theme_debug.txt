import { useCallback, useEffect, useRef, useState } from 'react'

export type Theme = 'light' | 'dark'
export type ThemePreference = 'system' | 'light' | 'dark'
const KEY = 'rtu_theme'
const THEME_CHANGE_EVENT = 'rtu-theme-change'

function resolveSystemTheme(): Theme {
  return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light'
}

function readPreference(): ThemePreference {
  try {
    const raw = localStorage.getItem(KEY)
    if (raw === 'system' || raw === 'light' || raw === 'dark') return raw
    return 'system'
  } catch {
    return 'system'
  }
}

function resolveTheme(pref: ThemePreference): Theme {
  if (pref === 'system') return resolveSystemTheme()
  return pref
}

function applyThemeColor(theme: Theme) {
  const meta = document.querySelector('meta[name="theme-color"]')
  if (meta) {
    meta.setAttribute('content', theme === 'dark' ? '#050505' : '#0e2f5a')
  }
}

/**
 * Light / dark mode with System, Light and Dark preferences.
 *
 * The choice is saved in the browser under key `rtu_theme`. When the key is
 * missing the preference is `'system'`, which resolves to light or dark based
 * on `prefers-color-scheme`. index.html applies the saved theme before React
 * loads so there is no flash.
 *
 * Multiple components can call useTheme() and they all stay in sync:
 * toggling the theme dispatches a window 'rtu-theme-change' event, and a
 * `storage` listener keeps open tabs in sync when `rtu_theme` changes.
 *
 * `<meta name="theme-color">` follows the active theme (uses the sticky navy
 * header colors so the mobile browser bar matches it).
 */
export function useTheme() {
  const [preference, setPreference] = useState<ThemePreference>(readPreference)
  const [theme, setTheme] = useState<Theme>(() => resolveTheme(preference))

  const systemMedia = useRef<MediaQueryList | null>(null)
  const toggleTimer = useRef<number | null>(null)
  const isFirstPref = useRef(true)

  // Apply the resolved theme to the DOM, meta tag and broadcast to other tabs.
  useEffect(() => {
    document.documentElement.dataset.theme = theme
    applyThemeColor(theme)
    window.dispatchEvent(
      new CustomEvent(THEME_CHANGE_EVENT, { detail: { theme } }),
    )
  }, [theme])

  // Save preference to localStorage only when the user explicitly changes it
  // (toggle / setPreference), not on first load. A missing key means 'system'.
  useEffect(() => {
    if (isFirstPref.current) {
      isFirstPref.current = false
      return
    }
    try {
      localStorage.setItem(KEY, preference)
    } catch {
      /* private mode: just skip saving */
    }
    window.dispatchEvent(
      new CustomEvent(THEME_CHANGE_EVENT, { detail: { theme } }),
    )
  }, [preference, theme])

  // Listen for theme changes dispatched from other components/tabs.
  useEffect(() => {
    const onThemeChange = (e: CustomEvent) => {
      const next = (e.detail as { theme?: Theme })?.theme
      if (next && next !== theme) {
        setTheme(next)
      }
    }
    window.addEventListener(
      THEME_CHANGE_EVENT as unknown as string,
      onThemeChange as EventListener,
    )
    return () => {
      window.removeEventListener(
        THEME_CHANGE_EVENT as unknown as string,
        onThemeChange as EventListener,
      )
    }
  }, [theme])

  // Keep `theme` in sync with the OS when preference is 'system'.
  useEffect(() => {
    const resolved = resolveTheme(preference)
    if (resolved !== theme) setTheme(resolved)

    if (preference === 'system') {
      const mql = window.matchMedia('(prefers-color-scheme: dark)')
      systemMedia.current = mql
      const onChange = () => {
        setTheme(resolveSystemTheme())
      }
      mql.addEventListener('change', onChange)
      return () => {
        mql.removeEventListener('change', onChange)
        systemMedia.current = null
      }
    }
    return undefined
  }, [preference, theme])

  // Sync across tabs via the storage event.
  useEffect(() => {
    const onStorage = (e: StorageEvent) => {
      if (e.key !== KEY) return
      const next = readPreference()
      if (next !== preference) setPreference(next)
    }
    window.addEventListener('storage', onStorage)
    return () => {
      window.removeEventListener('storage', onStorage)
    }
  }, [preference])

  // Cycle: system -> light -> dark -> system.
  // When on 'system' and the OS is currently dark, switch to 'light' first so
  // the click always produces a visible change.
  const toggle = useCallback(() => {
    setPreference((prev) => {
      if (prev === 'system') {
        return resolveSystemTheme() === 'dark' ? 'light' : 'dark'
      }
      if (prev === 'light') return 'dark'
      return 'system'
    })
  }, [])

  const setPref = useCallback((next: ThemePreference) => {
    setPreference(next)
  }, [])

  // theme-anim class for the toggle transition (skip when reduced motion).
  useEffect(() => {
    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    if (!reduced) {
      document.documentElement.classList.add('theme-anim')
      toggleTimer.current = window.setTimeout(() => {
        document.documentElement.classList.remove('theme-anim')
      }, 350)
    }
    return () => {
      if (toggleTimer.current !== null) {
        window.clearTimeout(toggleTimer.current)
        toggleTimer.current = null
      }
      document.documentElement.classList.remove('theme-anim')
    }
  }, [preference, theme])

  return {
    preference,
    theme,
    toggle,
    setPreference: setPref,
  }
}

import { useTheme } from '@/common/hooks/useTheme'
import { Icon } from '@/common/components/Icon'
import './ThemeToggle.css'

/**
 * THEME TOGGLE
 * Button that cycles through System -> Light -> Dark -> System.
 * Sits on the navy header (PubTop / Topbar) and keeps its navy look
 * in both themes.
 */
export function ThemeToggle() {
  const { preference, toggle } = useTheme()

  const labels: Record<'system' | 'light' | 'dark', string> = {
    system: 'Theme: System (click to switch)',
    light: 'Theme: Light (click to switch)',
    dark: 'Theme: Dark (click to switch)',
  }

  const titles: Record<'system' | 'light' | 'dark', string> = {
    system: 'System',
    light: 'Light',
    dark: 'Dark',
  }

  const isSystem = preference === 'system'
  const iconName: 'sun' | 'moon' | null = isSystem ? null : preference === 'light' ? 'sun' : 'moon'

  return (
    <button
      type="button"
      className="theme-toggle"
      onClick={toggle}
      aria-label={labels[preference]}
      title={titles[preference]}
    >
      {isSystem ? (
        <svg
          viewBox="0 0 24 24"
          width="16"
          height="16"
          fill="none"
          stroke="currentColor"
          strokeWidth={2}
          strokeLinecap="round"
          strokeLinejoin="round"
        >
          <rect x={2} y={7} width={20} height={12} rx={1} />
          <path d="M12 19v3" />
          <path d="M10 22h4" />
        </svg>
      ) : (
        <Icon name={iconName} size={16} />
      )}
    </button>
  )
}

import { useState, useCallback, useEffect, useRef } from 'react'
import logoImg from '@/assets/logo-removebg-preview.png'
import { navigate, useRoute } from '@/common/router/router'
import './PubTop.css'

export type PubTopSection = 'about' | 'insights' | 'help'

interface PubTopProps {
onSection?: (section: PubTopSection) => void
onNavigate?: (route: string) => void
}

export function PubTop({ onSection, onNavigate }: PubTopProps = {}) {
const { path } = useRoute()
// null = no tab highlighted (at top / hero / after clicking logo)
const [activeSection, setActiveSection] = useState<PubTopSection | null>(null)
const [dropdownOpen, setDropdownOpen] = useState(false)
const [isMobile, setIsMobile] = useState(window.innerWidth <= 900)
const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
// Prevents scroll spy from clearing the highlight right after a programmatic scroll
const isProgrammatic = useRef(false)

// Scroll spy: highlight the tab whose section is nearest the top of the viewport.
// When back at the hero (top of page), clear the highlight.
useEffect(() => {
    const handleScroll = () => {
    if (isProgrammatic.current) return

    const sections: { id: PubTopSection; elementId: string }[] = [
        { id: 'about', elementId: 'about' },
        { id: 'insights', elementId: 'insights' },
        { id: 'help', elementId: 'help' },
    ]
    let current: PubTopSection | undefined

    for (const { id, elementId } of sections) {
        const el = document.getElementById(elementId)
        if (el && el.getBoundingClientRect().top <= 120) {
        current = id
        }
    }

    setActiveSection(current ?? null)
    }

    handleScroll() // check on mount
    window.addEventListener('scroll', handleScroll, { passive: true })
    return () => window.removeEventListener('scroll', handleScroll)
}, [])

// Track mobile viewport (matches the max-width: 900px breakpoint in PubTop.css)
useEffect(() => {
    const onResize = () => setIsMobile(window.innerWidth <= 900)
    window.addEventListener('resize', onResize)
    return () => window.removeEventListener('resize', onResize)
}, [])

// Scroll to a section stored in sessionStorage after navigating to the landing page
// from a non-landing page (e.g. student-sellers or store guest pages).
useEffect(() => {
    const isLanding = path === '' || path === 'landing'
    if (!isLanding) return

    const target = sessionStorage.getItem('rtu:scroll-to-section')
    if (!target) return

    sessionStorage.removeItem('rtu:scroll-to-section')

    const timer = setTimeout(() => {
    isProgrammatic.current = true
    setActiveSection(target as PubTopSection)
    const el = document.getElementById(target)
    el?.scrollIntoView({ behavior: 'smooth', block: 'start' })
    setTimeout(() => (isProgrammatic.current = false), 800)
    }, 100)

    return () => clearTimeout(timer)
}, [path])

// Close the About Us dropdown when clicking outside the nav
useEffect(() => {
    if (!dropdownOpen) return
    const onDocClick = (e: MouseEvent) => {
    const target = e.target as Node
    const nav = document.querySelector('.pub-top nav')
    if (nav && !nav.contains(target)) {
        setDropdownOpen(false)
    }
    }
    window.addEventListener('click', onDocClick)
    return () => window.removeEventListener('click', onDocClick)
}, [dropdownOpen])

// Close mobile menu when clicking outside
useEffect(() => {
    if (!mobileMenuOpen) return
    const onDocClick = (e: MouseEvent) => {
    const target = e.target as Node
    const btn = document.querySelector('.mobile-menu-btn')
    const panel = document.querySelector('.mobile-menu-panel')
    if (btn && !btn.contains(target) && panel && !panel.contains(target)) {
        setMobileMenuOpen(false)
    }
    }
    window.addEventListener('click', onDocClick)
    return () => window.removeEventListener('click', onDocClick)
}, [mobileMenuOpen])

// Logo: scroll smoothly to top and clear the highlight
const handleLogo = useCallback(() => {
    isProgrammatic.current = true
    // Set the hash directly (useRoute picks it up via hashchange) so the
    // router's instant scroll doesn't override the smooth scroll below.
    window.location.hash = '#/'
    window.scrollTo({ top: 0, behavior: 'smooth' })
    setActiveSection(null)
    setTimeout(() => (isProgrammatic.current = false), 800)
}, [])

// Auth buttons: go to their proper pages.
// On mobile, flag the next page to scroll down after 3 seconds (the timer
// can't live in PubTop because this component unmounts on navigation).
const handleAuth = useCallback((route: string) => {
    onNavigate?.(route)

    if (isMobile && (route === 'login' || route === 'register')) {
        sessionStorage.setItem('rtu:scroll-after-login', 'true')
    }

    navigate(route)
}, [isMobile, onNavigate])

// Nav tabs: highlight clicked tab and scroll to its section.
// If not on the landing page, navigate there first and store the target
// section in sessionStorage so the landing page can scroll to it on mount.
const handleSection = useCallback((section: PubTopSection) => {
    const isLanding = path === '' || path === 'landing'
    if (!isLanding) {
    sessionStorage.setItem('rtu:scroll-to-section', section)
    navigate('')
    setDropdownOpen(false)
    return
    }
    isProgrammatic.current = true
    onSection?.(section)
    setActiveSection(section)
    const el = document.getElementById(section)
    el?.scrollIntoView({ behavior: 'smooth', block: 'start' })
    setTimeout(() => (isProgrammatic.current = false), 800)
}, [path, onSection, navigate])

// Only show the About Us dropdown as active when on the landing page and a section is open.
// This prevents stale state from showing it as active on Student Sellers / Official RTU Store.
const isLanding = path === '' || path === 'landing'
const isDropdownActive =
    isLanding && (activeSection === 'about' || activeSection === 'insights' || activeSection === 'help')

return (
    <header className="pub-top">
    <button
        type="button"
        className="logo"
        onClick={handleLogo}
        aria-label="RTU Tindahan home"
    >
        <img className="logo-img" src={logoImg} alt="RTU Tindahan Campus Marketplace" />
    </button>

    <nav className="pub-top-nav" aria-label="Sections">
        {/* About Us dropdown: contains About Us (trigger), Insights, Need Help */}
        <div className="nav-dropdown" onClick={(e) => e.stopPropagation()}>
        <button
            type="button"
            className={`nav-tab${isDropdownActive ? ' on' : ''} dropdown-trigger`}
            data-act="gnav"
            data-s="about"
            aria-current={isDropdownActive ? 'location' : undefined}
            aria-haspopup="true"
            aria-expanded={dropdownOpen}
            onClick={() => setDropdownOpen(!dropdownOpen)}
        >
            About Us<span className="dropdown-arrow"></span>
        </button>
        {dropdownOpen && (
            <div className="nav-dropdown-menu" role="menu">
            <button
                type="button"
                className="nav-dropdown-item"
                role="menuitem"
                onClick={() => {
                setDropdownOpen(false)
                handleSection('about')
                }}
            >
                About Us
            </button>
            <button
                type="button"
                className="nav-dropdown-item"
                role="menuitem"
                onClick={() => {
                setDropdownOpen(false)
                handleSection('insights')
                }}
            >
                Insights
            </button>
            <button
                type="button"
                className="nav-dropdown-item"
                role="menuitem"
                onClick={() => {
                setDropdownOpen(false)
                handleSection('help')
                }}
            >
                Need Help?
            </button>
            </div>
        )}
        </div>

        {/* Guest-facing marketplace buttons */}
        <button
        type="button"
        className={`nav-tab${path === 'student-sellers' ? ' on' : ''}`}
        aria-current={path === 'student-sellers' ? 'location' : undefined}
        onClick={() => {
        setDropdownOpen(false)
        navigate('student-sellers')
        }}
        >
        Student Sellers
        </button>
        <button
        type="button"
        className={`nav-tab${path === 'store' ? ' on' : ''}`}
        aria-current={path === 'store' ? 'location' : undefined}
        onClick={() => {
        setDropdownOpen(false)
        navigate('store')
        }}
        >
        Official RTU Store
        </button>
    </nav>

    <span className="pub-top-spacer" />

    <div className="pub-top-actions">
        {isMobile && (
        <button
            type="button"
            className="mobile-menu-btn"
            aria-expanded={mobileMenuOpen}
            onClick={() => setMobileMenuOpen(!mobileMenuOpen)}
        >
            â˜° Menu
        </button>
        )}
        {!isMobile && (
        <>
        <button
            type="button"
            className="btn ghost sm"
            onClick={() => handleAuth('login')}
            >
            Sign in
            </button>
        <button
            type="button"
            className="btn sm"
            onClick={() => handleAuth('register')}
            >
            Create account
            </button>
        </>
        )}
    </div>

    {isMobile && mobileMenuOpen && (
        <div className="mobile-menu-panel">
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            handleSection('about')
            }}
        >
            About Us
        </button>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            handleSection('insights')
            }}
        >
            Insights
        </button>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            handleSection('help')
            }}
        >
            Need Help?
        </button>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            navigate('student-sellers')
            }}
        >
            Student Sellers
        </button>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            navigate('store')
            }}
        >
            Official RTU Store
        </button>
        <div style={{ padding: '8px 16px', borderTop: '1px solid var(--line)', marginTop: 8, fontSize: 11, color: 'var(--ink)', textTransform: 'uppercase', letterSpacing: '0.5px' }}>
            Account
        </div>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            handleAuth('login')
            }}
        >
            Sign in
        </button>
        <button
            type="button"
            className="mobile-menu-item"
            onClick={() => {
            setMobileMenuOpen(false)
            handleAuth('register')
            }}
        >
            Create account
        </button>
        </div>
    )}
    </header>
    )
}
