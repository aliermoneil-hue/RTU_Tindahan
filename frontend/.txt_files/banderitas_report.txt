import { useId } from 'react'

import './Banderitas.css'

export type BanderitasVariant =
  | 'corner-left'
  | 'corner-right'
  | 'canopy'
  | 'garland'
  | 'mini'

interface BanderitasProps {
  variant?: BanderitasVariant
  opacity?: number
  className?: string
  surface?: 'dark' | 'light'
  compact?: boolean
}

/** Brand colors for dark (navy) surfaces — cream flag pops on dark. */
const DARK_SURFACE_COLORS = ['#f4b63a', '#c84b4b', '#2563eb', '#fffaf0']
/** Brand colors for light/white surfaces — navy flag replaces cream so no white-on-white. */
const LIGHT_SURFACE_COLORS = ['#f4b63a', '#c84b4b', '#2563eb', '#1a2f57']

/** Build pennant polygons whose top edge sits flush on a quadratic Bezier curve. */
function buildPennants(
  p0: [number, number],
  p1: [number, number],
  p2: [number, number],
  count: number,
  flagWidth: number,
  flagHeight: number,
  colors: string[],
) {
  const pt = (t: number): [number, number] => {
    const u = 1 - t
    return [
      u * u * p0[0] + 2 * u * t * p1[0] + t * t * p2[0],
      u * u * p0[1] + 2 * u * t * p1[1] + t * t * p2[1],
    ]
  }
  const tan = (t: number): [number, number] => [
    2 * (1 - t) * (p1[0] - p0[0]) + 2 * t * (p2[0] - p1[0]),
    2 * (1 - t) * (p1[1] - p0[1]) + 2 * t * (p2[1] - p1[1]),
  ]

  return Array.from({ length: count }, (_, i) => {
    const t = (i + 0.5) / count
    const [cx, cy] = pt(t)
    const [tx, ty] = tan(t)
    const len = Math.hypot(tx, ty) || 1
    const ux = tx / len
    const uy = ty / len
    let nx = -uy
    let ny = ux
    if (ny < 0) {
      nx = -nx
      ny = -ny
    }
    const half = flagWidth / 2
    const x1 = cx - ux * half
    const y1 = cy - uy * half
    const x2 = cx + ux * half
    const y2 = cy + uy * half
    const x3 = cx + nx * flagHeight
    const y3 = cy + ny * flagHeight
    return {
      points: `${x1.toFixed(1)},${y1.toFixed(1)} ${x2.toFixed(1)},${y2.toFixed(1)} ${x3.toFixed(1)},${y3.toFixed(1)}`,
      fill: colors[i % colors.length],
    }
  })
}

/** Build bulb positions along a quadratic Bezier strand (night-market string lights). */
function buildBulbs(
  p0: [number, number],
  p1: [number, number],
  p2: [number, number],
  count: number,
) {
  const pt = (t: number): [number, number] => {
    const u = 1 - t
    return [
      u * u * p0[0] + 2 * u * t * p1[0] + t * t * p2[0],
      u * u * p0[1] + 2 * u * t * p1[1] + t * t * p2[1],
    ]
  }
  return Array.from({ length: count }, (_, i) => {
    const t = (i + 0.5) / count
    const [cx, cy] = pt(t)
    return { cx, cy, delay: `${((i * 0.43) % 2.6).toFixed(2)}s` }
  })
}

/** Render one curved strand with both pennants (light mode) and lights (dark mode). */
function Strand({
  p0,
  p1,
  p2,
  count,
  flagWidth,
  flagHeight,
  opacity = 1,
  colors = DARK_SURFACE_COLORS,
  glowId,
  compact = false,
}: {
  p0: [number, number]
  p1: [number, number]
  p2: [number, number]
  count: number
  flagWidth: number
  flagHeight: number
  opacity?: number
  colors?: string[]
  glowId: string
  compact?: boolean
}) {
  const pennants = buildPennants(p0, p1, p2, count, flagWidth, flagHeight, colors)
  const bulbs = buildBulbs(p0, p1, p2, count)
  return (
    <g style={{ opacity }}>
      <path
        className="banderitas-string"
        d={`M ${p0[0]},${p0[1]} Q ${p1[0]},${p1[1]} ${p2[0]},${p2[1]}`}
        fill="none"
        stroke="rgba(225, 195, 145, 0.85)"
        strokeWidth="1.8"
        strokeLinecap="round"
        vector-effect="non-scaling-stroke"
      />
      <g className="banderitas-pennants">
        {pennants.map((p, i) => (
          <polygon
            key={`p-${i}`}
            className="banderitas-flag"
            data-i={i}
            points={p.points}
            fill={p.fill}
            stroke="rgba(0, 0, 0, 0.08)"
            strokeWidth="0.5"
            vector-effect="non-scaling-stroke"
            style={{ animationDelay: `${(i % 5) * 0.28}s` }}
          />
        ))}
      </g>
      <g className="banderitas-lights">
        {bulbs.map((b, i) => (
          <g key={`b-${i}`}>
            <line
              x1={b.cx}
              y1={b.cy}
              x2={b.cx}
              y2={b.cy + 5}
              className="banderitas-socket"
            />
            <circle
              cx={b.cx}
              cy={b.cy + 8}
              r={compact ? 11 : 14}
              fill={`url(#bulb-glow-${glowId})`}
              className="banderitas-halo"
              style={{ animationDelay: b.delay }}
            />
            <circle
              cx={b.cx}
              cy={b.cy + 8}
              r={compact ? 2.6 : 3.4}
              className="banderitas-bulb"
              style={{ animationDelay: b.delay }}
            />
          </g>
        ))}
      </g>
    </g>
  )
}

export function Banderitas({
  variant = 'garland',
  opacity = 0.9,
  className,
  surface = 'dark',
  compact = false,
}: BanderitasProps) {
  const s = opacity
  const lightCls = surface === 'light' ? ' banderitas--light' : ''
  const compactCls = compact ? ' banderitas--compact' : ''
  const cls = `banderitas banderitas--${variant}${className ? ` ${className}` : ''}${lightCls}${compactCls}`
  const colors = surface === 'light' ? LIGHT_SURFACE_COLORS : DARK_SURFACE_COLORS
  const glowId = useId().replace(/:/g, '')

  const defs = (
    <defs>
      <radialGradient id={`bulb-glow-${glowId}`}>
        <stop offset="0%" stopColor="#fff4c2" stopOpacity={0.95} />
        <stop offset="30%" stopColor="#f5b942" stopOpacity={0.55} />
        <stop offset="70%" stopColor="#e59a24" stopOpacity={0.14} />
        <stop offset="100%" stopColor="#e59a24" stopOpacity={0} />
      </radialGradient>
    </defs>
  )

  if (variant === 'corner-left' && compact) {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 150 65"
        preserveAspectRatio="none"
      >
        {defs}
        <Strand p0={[-4, 46]} p1={[66, 34]} p2={[136, -4]} count={5} flagWidth={18} flagHeight={20} colors={colors} glowId={glowId} compact />
      </svg>
    )
  }

  if (variant === 'corner-left') {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 260 120"
        preserveAspectRatio="none"
      >
        {defs}
        <Strand p0={[-6, 96]} p1={[115, 68]} p2={[250, -6]} count={8} flagWidth={22} flagHeight={26} colors={colors} glowId={glowId} />
        <Strand p0={[-6, 48]} p1={[70, 38]} p2={[145, -6]} count={5} flagWidth={18} flagHeight={21} opacity={0.72} colors={colors} glowId={glowId} />
      </svg>
    )
  }

  if (variant === 'corner-right' && compact) {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 150 65"
        preserveAspectRatio="none"
      >
        {defs}
        <Strand p0={[14, -4]} p1={[84, 34]} p2={[154, 46]} count={5} flagWidth={18} flagHeight={20} colors={colors} glowId={glowId} compact />
      </svg>
    )
  }

  if (variant === 'corner-right') {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 260 130"
        preserveAspectRatio="none"
      >
        {defs}
        <Strand p0={[10, -6]} p1={[140, 76]} p2={[266, 112]} count={9} flagWidth={22} flagHeight={26} colors={colors} glowId={glowId} />
        <Strand p0={[110, -6]} p1={[190, 42]} p2={[266, 54]} count={5} flagWidth={18} flagHeight={21} opacity={0.72} colors={colors} glowId={glowId} />
      </svg>
    )
  }

  if (variant === 'canopy') {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 1200 170"
        preserveAspectRatio="xMidYMin slice"
      >
        {defs}
        <Strand p0={[-10, 115]} p1={[240, 65]} p2={[520, -6]} count={13} flagWidth={24} flagHeight={26} colors={colors} glowId={glowId} />
        <Strand p0={[640, -6]} p1={[940, 85]} p2={[1210, 135]} count={14} flagWidth={24} flagHeight={26} colors={colors} glowId={glowId} />
        <Strand p0={[-10, 18]} p1={[600, 68]} p2={[1210, 12]} count={24} flagWidth={20} flagHeight={20} opacity={0.55} colors={colors} glowId={glowId} />
      </svg>
    )
  }

  if (variant === 'garland') {
    return (
      <svg
        className={cls}
        aria-hidden="true"
        style={{ opacity: s }}
        viewBox="0 0 1200 48"
      >
        {defs}
        <Strand p0={[0, 2]} p1={[200, 24]} p2={[400, 4]} count={11} flagWidth={22} flagHeight={18} colors={colors} glowId={glowId} />
        <Strand p0={[400, 4]} p1={[600, 24]} p2={[800, 4]} count={11} flagWidth={22} flagHeight={18} colors={colors} glowId={glowId} />
        <Strand p0={[800, 4]} p1={[1000, 24]} p2={[1200, 2]} count={11} flagWidth={22} flagHeight={18} colors={colors} glowId={glowId} />
      </svg>
    )
  }

  // mini: compact 5-flag curved accent
  return (
    <svg
      className={cls}
      aria-hidden="true"
      style={{ opacity: s }}
      viewBox="0 0 140 32"
      preserveAspectRatio="none"
    >
      {defs}
      <Strand p0={[4, 4]} p1={[70, 18]} p2={[136, 4]} count={5} flagWidth={18} flagHeight={16} colors={colors} glowId={glowId} />
    </svg>
  )
}

/**

 */
export const LOADER_TIMING = {
  // Anti-flicker & readiness thresholds (ms)

  slowLoadThresholdMs: 8000, // Show "Taking longer than usual..." + Retry

  // Full intro build (~2.0s total)

    tileDropDuration: 0.32,

    exitDuration: 0.45,
  },

  // Compact variant (~0.9s build, ~0.3s exit) for repeat loads in same session
  compact: {

    exitDuration: 0.3,
  },

  // Fast-forward multiplier when app becomes ready mid-build (legacy fallback)
  fastForwardTimeScale: 3.5,

  // Adaptive exit durations based on actual load speed (seconds)
  adaptiveExitFast: 0.24,   // < 400ms load

  // Calm idle shine sweep interval (seconds)
  idleShineDuration: 0.75,

const SESSION_KEY = "rtu_tindahan_intro_seen";

export type BrandLoaderVariant = "full" | "compact" | "auto";

export interface BrandLoaderProps {
  /** True when the app/page data is still loading */
  isLoading: boolean;
  /** "auto" plays full on first session visit, compact on subsequent boots */
  variant?: BrandLoaderVariant;
  /** Optional retry callback for the 8s slow-load fallback */
  onRetry?: () => void;
  /** Called after the exit animation finishes and the loader unmounts */
  onExited?: () => void;
  /** GSAP timeScale multiplier for build & exit (1 = normal, 2 = 2x faster, etc.) */
  timeScale?: number;

function markIntroSeen(): void {
  try {
    window.sessionStorage.setItem(SESSION_KEY, "1");
  } catch {
    // Ignore storage errors in restricted/private contexts
  }

export const BrandLoader: React.FC<BrandLoaderProps> = ({
  isLoading,
  variant = "auto",
  onRetry,
  onExited,
  timeScale = 1,
  immediate = false,
}) => {
  const [shouldRender, setShouldRender] = useState<boolean>(false);
  const [isExiting, setIsExiting] = useState<boolean>(false);
  const [isSlowLoad, setIsSlowLoad] = useState<boolean>(false);
  const { theme } = useTheme()

  const overlayRef = useRef<HTMLDivElement | null>(null);
  const stageRef = useRef<HTMLDivElement | null>(null);
  const buildTlRef = useRef<gsap.core.Timeline | null>(null);
  const idleTlRef = useRef<gsap.core.Timeline | null>(null);
  const buildCompletedRef = useRef<boolean>(false);
  const exitStartedRef = useRef<boolean>(false);
  const isLoadingRef = useRef<boolean>(isLoading);
  /** Recorded the moment shouldRender becomes true — used to compute elapsed load time. */
  const loadStartMsRef = useRef<number>(0);
  /** True on the very first render. On initial mount with isLoading=true we skip
   * the anti-flicker delay so the brand loader always shows on a fresh page load. */
  const initialMountRef = useRef(true);
  /** Prop refs — updated via effects so the build effect (which depends only on
   * shouldRender) always reads the latest props without re-running. */
  const variantRef = useRef<BrandLoaderVariant>(variant);
  const onExitedRef = useRef<(() => void) | undefined>(onExited);
  const timeScaleRef = useRef<number>(timeScale);
  /** Adaptive exit duration in seconds, set by accelerateToFinish(). */
  const adaptiveExitSecRef = useRef<number>(0.32);

  useEffect(() => {
    isLoadingRef.current = isLoading;
  }, [isLoading]);

  // Keep prop refs in sync so the build effect (which depends only on shouldRender)
  // always reads the latest props without re-running.
  useEffect(() => {
    variantRef.current = variant;
  }, [variant]);
  useEffect(() => {
    onExitedRef.current = onExited;
  }, [onExited]);
  useEffect(() => {
    timeScaleRef.current = timeScale;
  }, [timeScale]);

  // 1. Anti-flicker gate: don't mount overlay if loading finishes in < 150ms.
  // On initial mount with isLoading=true we show immediately — a full page reload
  // should always play the brand intro. When `immediate` is true (login flow) we
  // also bypass the delay so fast logins don't delay or skip the animation.
  // Subsequent loading states (route changes, data refetches) still get the
  // 150ms debounce to avoid flicker.
  useEffect(() => {
    if (!isLoading) return;
    if (shouldRender) return;

    if (initialMountRef.current || immediate) {
      loadStartMsRef.current = Date.now();
      setShouldRender(true);
      initialMountRef.current = false;
      return;
    }

    const timer = window.setTimeout(() => {
      if (isLoadingRef.current) {
        setShouldRender(true);
      }
    }, LOADER_TIMING.antiFlickerDelayMs);

    return () => window.clearTimeout(timer);
  }, [isLoading, shouldRender, immediate]);

  // 2. Slow-load timer (8s)
  useEffect(() => {
    if (!shouldRender || !isLoading) {
      setIsSlowLoad(false);
      return;
    }

    const slowTimer = window.setTimeout(() => {
      if (isLoadingRef.current) {
        setIsSlowLoad(true);
      }
    }, LOADER_TIMING.slowLoadThresholdMs);

    return () => window.clearTimeout(slowTimer);
  }, [shouldRender, isLoading]);

  // 3. Scroll lock + aria-busy on #root while visible
  useEffect(() => {
    if (!shouldRender) return;

    const rootEl = document.getElementById("root");
    document.body.style.overflow = "hidden";
    rootEl?.setAttribute("aria-busy", "true");

    return () => {
      // Clear overflow so CSS (and the app's normal scrolling) takes over.
      document.body.style.removeProperty("overflow");
      rootEl?.removeAttribute("aria-busy");
      // Always clear the inline navy background so the app's CSS background
      // (var(--bg)) shows through even if the exit animation never ran.
      document.body.style.removeProperty("background-color");
      document.body.style.removeProperty("margin");
      document.body.style.removeProperty("height");
    };
  }, [shouldRender]);

  // 7. Safety net: if the loader stays visible for more than 10 seconds
  //    (e.g. auth/restore or listings fetch is hung), force the exit so the
  //    user is never trapped behind a blank navy overlay.
  useEffect(() => {
    if (!shouldRender || !isLoading) return;
    const safetyTimer = window.setTimeout(() => {
      runExit(0.2);
    }, 10000);
    return () => window.clearTimeout(safetyTimer);
  }, [shouldRender, isLoading]);

  // 4. Trigger exit animation helper
  const runExit = (adaptiveExitDuration?: number) => {
    if (exitStartedRef.current || !overlayRef.current || !stageRef.current) return;
    exitStartedRef.current = true;
    setIsExiting(true);
    idleTlRef.current?.kill();
    markIntroSeen();

    // Clear <body> inline navy styles immediately so the app's normal
    // background (var(--bg)) is already visible when the overlay fades out.
    // Doing this at the start of exit (not at onComplete) prevents a flash
    // of navy between the overlay disappearing and the styles clearing.
    const bodyEl = document.body;
    bodyEl.style.removeProperty("background-color");
    bodyEl.style.removeProperty("margin");
    bodyEl.style.removeProperty("height");
    bodyEl.style.removeProperty("overflow");

    const prefersReduced = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;

    let exitDuration: number;
    if (prefersReduced) {
      exitDuration = LOADER_TIMING.reducedMotionFade;
    } else if (adaptiveExitDuration !== undefined) {
      exitDuration = adaptiveExitDuration;
    } else {
      exitDuration = adaptiveExitSecRef.current || 0.32;
    }

    const tl = gsap.timeline({
      onComplete: () => {
        setShouldRender(false);
        setIsExiting(false);
        exitStartedRef.current = false;
        buildCompletedRef.current = false;
        onExited?.();
      },
    });

    if (prefersReduced) {
      tl.to(overlayRef.current, { opacity: 0, duration: exitDuration, ease: "power1.out" });
    } else {
      tl.to(stageRef.current, {
        y: -24,
        scale: 1.05,
        opacity: 0,
        duration: exitDuration * 0.8,
        ease: "power2.in",
      }).to(
        overlayRef.current,
        {
          opacity: 0,
          duration: exitDuration * 0.75,
          ease: "power2.out",
        },
        `-=${exitDuration * 0.45}`
      );
    }
  };

  // 5. GSAP Build & Idle Timeline (StrictMode-safe via gsap.context)
  //    Depends ONLY on [shouldRender] so prop changes (variant/timeScale/onExited)
  //    when isSigningIn flips never trigger ctx.revert() or kill the running timeline.
  useLayoutEffect(() => {
    if (!shouldRender || !overlayRef.current) return;

    // Resolve variant once for this run — never re-resolve on prop changes.
    const currentVariant =
      variantRef.current !== "auto"
        ? variantRef.current
        : (() => {
            try {
              return window.sessionStorage.getItem(SESSION_KEY)
                ? "compact"
                : "full";
            } catch {
              return "full";
            }
          })();
    const currentTimeScale = timeScaleRef.current;

    const prefersReduced = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;

    const ctx = gsap.context(() => {
      if (prefersReduced) {
        // Reduced motion: show all elements statically, then fade the overlay in
        gsap.set([
          ".bl-tindahan",
          ".bl-tagline-text",
          ".bl-tagline-line-left",
          ".bl-tagline-line-right",
          ".bl-tile",
          ".bl-letter",
          ".bl-cursor",
          ".bl-tag-group",
          ".bl-shine",
        ], { opacity: 1, scaleX: 1, y: 0, x: 0, skewX: 0, rotation: 0, scale: 1, transformOrigin: "center center" });
        gsap.fromTo(
          overlayRef.current,
          { opacity: 0 },
          {
            opacity: 1,
            duration: LOADER_TIMING.reducedMotionFade,
            onComplete: () => {
              buildCompletedRef.current = true;
              if (!isLoadingRef.current) {
                runExit(adaptiveExitSecRef.current);
              }
            },
          }
        );
        return;
      }

      const cfg = LOADER_TIMING.full;

      // Initial hidden states (transform & opacity only)
      gsap.set(".bl-tindahan", { x: -70, skewX: -12, opacity: 0 });
      gsap.set(".bl-tagline-text", { opacity: 0, scale: 0.94, transformOrigin: "center center" });
      gsap.set(".bl-tagline-line-left", { scaleX: 0, transformOrigin: "right center" });
      gsap.set(".bl-tagline-line-right", { scaleX: 0, transformOrigin: "left center" });
      gsap.set(".bl-tile", { y: -32, scaleY: 0.75, opacity: 0, transformOrigin: "center bottom" });
      gsap.set(".bl-awning-group", { transformOrigin: "center bottom" });
      gsap.set(".bl-letter", { y: -70, opacity: 0, transformOrigin: "center bottom" });
      gsap.set(".bl-cursor", { scale: 0, opacity: 0, transformOrigin: "20% 20%" });
      gsap.set(".bl-tag-group", { rotation: -28, opacity: 0, transformOrigin: "100% 0%" });
      gsap.set(".bl-shine", { x: -460, opacity: 0 });

      const tl = gsap.timeline({
        onComplete: () => {
          buildCompletedRef.current = true;
          if (!isLoadingRef.current) {
            runExit(adaptiveExitSecRef.current);
          } else {
            // Calm idle state: soft shine sweep across the awning every ~1.6s
            idleTlRef.current = gsap
              .timeline({ repeat: -1, repeatDelay: LOADER_TIMING.idleShineRepeatDelay })
              .fromTo(
                ".bl-shine",
                { x: -460, opacity: 0 },
                {
                  x: 520,
                  opacity: 0.45,
                  duration: LOADER_TIMING.idleShineDuration,
                  ease: "power1.inOut",
                }
              )
              .to(".bl-shine", { opacity: 0, duration: 0.15 }, "-=0.15");
          }
        },
      });

      // STEP 1: TINDAHAN rushes in + tagline lines draw outward from center
      tl.to(".bl-tindahan", {
        x: 0,
        skewX: 0,
        opacity: 1,
        duration: cfg.tindahanDuration,
        ease: "power3.out",
      })
        .to(
          ".bl-tagline-text",
          { opacity: 1, scale: 1, duration: 0.3, ease: "power2.out" },
          `-=${cfg.tindahanDuration * 0.5}`
        )
        .to(
          [".bl-tagline-line-left", ".bl-tagline-line-right"],
          { scaleX: 1, opacity: 1, duration: cfg.taglineLinesDuration, ease: "power2.out" },
          `-=${cfg.tindahanDuration * 0.45}`
        );

      // STEP 2: 17 Awning tiles drop left-to-right with overshoot + squash
      tl.to(
        ".bl-tile",
        {
          y: 0,
          scaleY: 1,
          opacity: 1,
          duration: cfg.tileDropDuration,
          stagger: cfg.tileStagger,
          ease: "back.out(1.6)",
        },
        cfg.tilesOverlap
      ).to(
        ".bl-tag-group",
        {
          rotation: 0,
          opacity: 1,
          duration: cfg.tagSwingDuration,
          ease: "elastic.out(1, 0.45)",
        },
        "-=0.35"
      );

      // STEP 3: R drops first, lands on awning with bounce
      tl.to(".bl-letter-r", {
        y: 0,
        opacity: 1,
        duration: cfg.letterDropDuration,
        ease: "bounce.out",
      }, "+=0.1")
        .to(".bl-awning-group", {
          scaleY: 0.94,
          duration: 0.07,
          yoyo: true,
          repeat: 1,
          ease: "power1.inOut",
        }, "-=0.15")
        // T drops right after R lands
        .to(".bl-letter-t", {
          y: 0,
          opacity: 1,
          duration: cfg.letterDropDuration,
          ease: "bounce.out",
        }, "-=0.22")
        .to(".bl-awning-group", {
          scaleY: 0.94,
          duration: 0.07,
          yoyo: true,
          repeat: 1,
          ease: "power1.inOut",
        }, "-=0.15")
        // U drops right after T lands
        .to(".bl-letter-u", {
          y: 0,
          opacity: 1,
          duration: cfg.letterDropDuration,
          ease: "bounce.out",
        }, "-=0.22")
        .to(".bl-awning-group", {
          scaleY: 0.94,
          duration: 0.07,
          yoyo: true,
          repeat: 1,
          ease: "power1.inOut",
        }, "-=0.15");

      // STEP 4: Cursor arrow pops in and performs a single "click" press
      tl.to(
        ".bl-cursor",
        {
          scale: 1,
          opacity: 1,
          duration: cfg.cursorPopDuration,
          ease: "back.out(2)",
        },
        "-=0.1"
      ).to(".bl-cursor", {
        scale: 0.84,
        duration: cfg.cursorClickDuration * 0.5,
        yoyo: true,
        repeat: 1,
        ease: "power2.inOut",
      });

      // Network-aware starting speed: slow networks play slower, fast networks play at normal/compact speed
      const effType = (navigator as any).connection?.effectiveType;
      const isSlowNetwork =
        effType === "slow-2g" || effType === "2g" || effType === "3g";
      let startTimeScale: number;
      if (isSlowNetwork) {
        startTimeScale = 0.75;
      } else if (currentVariant === "compact") {
        startTimeScale = LOADER_TIMING.compact.timeScale;
      } else if (currentTimeScale && currentTimeScale !== 1) {
        startTimeScale = currentTimeScale;
      } else {
        startTimeScale = 1.0;
      }
      tl.timeScale(startTimeScale);

      buildTlRef.current = tl;
    }, overlayRef);

    return () => {
      buildTlRef.current = null;
      idleTlRef.current = null;
      ctx.revert();
    };
  }, [shouldRender]);

  // Accelerate the build timeline to finish in targetSec seconds,
  // computed from remaining time and actual load elapsed time.
  const accelerateToFinish = (tl: gsap.core.Timeline) => {
    const remaining = Math.max(0.05, tl.duration() - tl.time());
    const elapsed = Date.now() - loadStartMsRef.current;
    // Fast load finishes remaining steps briskly (~0.42s); slower load eases out (~0.58s)
    const targetSec = elapsed < 400 ? 0.42 : elapsed < 1200 ? 0.50 : 0.58;

    adaptiveExitSecRef.current = elapsed < 400 ? 0.24 : elapsed < 1200 ? 0.32 : 0.40;
    gsap.killTweensOf(tl);
    gsap.to(tl, {
      timeScale: neededScale,
      duration: 0.12,
      ease: "power2.out",
      overwrite: true,
    });
  };

  //    - While loading drags on, smoothly decelerate the build (calm, never frozen)
  //    - When loading finishes, fast-forward proportionally to how fast it actually was

  //      the exit so the overlay never traps the app under a blank navy screen.

import React, { useEffect, useRef, useState } from "react";
import "./BrandLoader.css";

const MINI_TILES = ["gold", "red", "white", "gold", "red"] as const;

export interface MiniLoaderProps {
  /** Controls whether loading is active (built-in 200ms delay & 400ms min duration) */
  isLoading?: boolean;
  /** "inline" for buttons, "section" for page/card content areas */
  variant?: "inline" | "section";
  size?: "sm" | "md" | "lg";
  label?: string;
  className?: string;
}

export const MiniLoader: React.FC<MiniLoaderProps> = ({
  isLoading = true,
  variant = "inline",
  size = "md",
  label,
  className = "",
}) => {
  // Inline button spinners show immediately; section loaders use 200ms anti-flicker delay
  const [visible, setVisible] = useState<boolean>(variant === "inline" && isLoading);
  const shownAtRef = useRef<number>(variant === "inline" && isLoading ? Date.now() : 0);

  useEffect(() => {
    if (variant === "inline") {
      setVisible(isLoading);
      return;
    }

    if (isLoading) {
      const showTimer = window.setTimeout(() => {
        shownAtRef.current = Date.now();
        setVisible(true);
      }, 200);
      return () => window.clearTimeout(showTimer);
    }

    if (visible) {
      const elapsed = Date.now() - shownAtRef.current;
      const remaining = Math.max(0, 400 - elapsed);
      const hideTimer = window.setTimeout(() => setVisible(false), remaining);
      return () => window.clearTimeout(hideTimer);
    }
  }, [isLoading, variant, visible]);

  if (!visible) return null;

  return (
    <span
      className={`mini-loader mini-loader--${variant} mini-loader--${size} ${className}`.trim()}
      role="status"
      aria-live="polite"
    >
      <span className="mini-loader-tiles" aria-hidden="true">
        {MINI_TILES.map((color, idx) => (
          <span
            key={idx}
            className={`mini-loader-tile mini-loader-tile--${color}`}
            style={{ animationDelay: `${idx * 90}ms` }}
          />
        ))}
      </span>
      {label ? (
        <span className="mini-loader-label">{label}</span>
      ) : (
        <span className="bl-sr-only">Loading…</span>
      )}
    </span>
  );
};

import { useEffect, useState } from "react";

interface UseAppReadyOptions {
  authLoading: boolean;
  isAuthenticated: boolean;
  listingsSettled: boolean;
  /** Safety timeout (ms) so a broken/slow catalogue request never traps the loader */
  listingsTimeoutMs?: number;
}

/**
 * First-load readiness rule:
 * 1. Auth session restore settled (/auth/me/)
 * 2. If signed in, first listings fetch settled OR 3s timeout elapsed
 * 3. Document fonts ready (or 1.5s fallback)
 */
export function useAppReady({
  authLoading,
  isAuthenticated,
  listingsSettled,
  listingsTimeoutMs = 3000,
}: UseAppReadyOptions): boolean {
  const [fontsReady, setFontsReady] = useState<boolean>(false);
  const [listingsTimedOut, setListingsTimedOut] = useState<boolean>(false);

  // Wait for fonts with a safe fallback
  useEffect(() => {
    let active = true;
    const fallback = window.setTimeout(() => {
      if (active) setFontsReady(true);
    }, 1500);

    if (document.fonts && document.fonts.ready) {
      document.fonts.ready.then(() => {
        if (active) setFontsReady(true);
      });
    } else {
      setFontsReady(true);
    }

    return () => {
      active = false;
      window.clearTimeout(fallback);
    };
  }, []);

  // 3s safety timeout for the initial listings fetch when signed in
  useEffect(() => {
    if (authLoading || !isAuthenticated || listingsSettled) return;
    const timer = window.setTimeout(() => setListingsTimedOut(true), listingsTimeoutMs);
    return () => window.clearTimeout(timer);
  }, [authLoading, isAuthenticated, listingsSettled, listingsTimeoutMs]);

  const dataReady = !isAuthenticated || listingsSettled || listingsTimedOut;
  return !authLoading && dataReady && fontsReady;
}
