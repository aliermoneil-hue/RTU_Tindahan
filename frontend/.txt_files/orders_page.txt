export function BaoProgressBar({ txn }: { txn: TransactionRecord }) {
  const p = baoProgress(txn)
  return (
    <ol className="bstep" aria-label="Request progress">
      {BAO_STEPS.map((s, i) => {
        let cls = i < p.idx ? 'done' : i === p.idx ? 'cur' : 'todo'
        let mark: React.ReactNode = i < p.idx ? <Icon name="check" size="sm" /> : String(i + 1)
        let label: string = s

        if (i === p.idx && p.cur === 'Claimed') {
          cls = 'done'
          mark = <Icon name="check" size="sm" />
        }
        if (i === p.idx && p.hold) {
          cls = 'hold'
          mark = 'II'
          label = 'On Hold'
        }
        if (i === p.idx && p.cancelled) {
          cls = 'stop'
          mark = 'X'
          label = 'Cancelled'
        }
        return (
          <li key={s} className={cls}>
            <span className="dot">{mark}</span>
            <span className="lbl">{label}</span>
          </li>
        )
      })}
    </ol>
  )
}

                      </button>
                    )}
                    {!selIsOfficial && (
                      <button
                        type="button"
                        className="btn ghost sm"
                        onClick={() =>
                          openModal({
                            kind: 'order-report',
                            orderId: sel.id,
                            role: 'buyer',
                            targetEmail: sel.sellerEmail,
                            targetName: sel.seller,
                            itemTitle: sel.title,
                          })
                        }
                      >
                        <Icon name="flag" size="sm" />
                        Report issue
                      </button>
                    )}
                    {sel && selCanCancel ? (
                      <button
                        type="button"
                        className="btn danger sm"
                        onClick={() =>
                          setOpenBox((cur) => (cur === 'cancel' ? '' : 'cancel'))
                        }
                      >
                        Cancel request
                      </button>
                    ) : sel && !selCanCancel ? (
                      <button
                        type="button"
                        className="btn danger sm disabled"
                        disabled
                        title="Cannot cancel once meetup is set. Message the seller or request a reschedule instead."
                      >
                        Cancel request
                      </button>
                    ) : null}
                  </div>

                  {!sellerAccepted && !selIsOfficial && (
                    <div className="note blue">
                      <Icon name="info" size="sm" />
                      <span>
                        Confirm {sel.fulfillment === 'delivery' ? 'delivery' : 'pickup'} unlocks once the seller accepts your request.
                      </span>
                    </div>
                  )}

                  {openBox === 'msg' && (
                    <form
                      className="card flat stack"
                      style={{ background: 'var(--bg)' }}
                      onSubmit={(e) => {
                        e.preventDefault()
                        if (sendOrderMessage(sel.id, msgText, true)) {
                          setMsgText('')
                          setOpenBox('')
                        }
                      }}
                    >
                      <b className="small">Message {sel.seller}</b>
                      <textarea
                        className="input"
                        id="omsg-t"
                        placeholder="Write a message about this meetup..."
                        value={msgText}
                        onChange={(e) => setMsgText(e.target.value)}
                      />
                      <div className="row" style={{ justifyContent: 'flex-end' }}>
                        <button
                          type="button"
                          className="btn ghost sm"
                          onClick={() => setOpenBox('')}
                        >
                          Close
                        </button>
                        <button type="submit" className="btn sm">
                          Send
                        </button>
                      </div>
                    </form>
                  )}

                  {!selIsOfficial && openBox === 'edit' && (
                    <form
                      className="card flat stack"
                      style={{ background: 'var(--bg)' }}
                      onSubmit={(e) => {
                        e.preventDefault()
                        proposeOrderMeetup(sel.id, editPlace, editTime)
                        setOpenBox('')
                      }}
                    >
                      <b className="small">Propose a new meetup</b>
                      <div className="grid2">
                        <div className="field">
                          <label htmlFor="oep">Location</label>
                          <select
                            className="input"
                            id="oep"
                            value={
                              editOthersMode || !presetPlaces.includes(editPlace)
                                ? 'Others'
                                : editPlace
                            }
                            onChange={(e) => {
                              const val = e.target.value
                              if (val === 'Others') {
                                setEditOthersMode(true)
                                if (presetPlaces.includes(editPlace)) setEditPlace('')
                              } else {
                                setEditOthersMode(false)
                                setEditPlace(val)
                              }
                            }}
                          >
                            {PLACES.map((pl) => (
                              <option key={pl} value={pl}>
                                {pl}
                              </option>
                            ))}
                          </select>
                          {(editOthersMode || !presetPlaces.includes(editPlace)) && (
                            <input
                              className="input"
                              style={{ marginTop: 8 }}
                              placeholder="Write your specific campus spot..."
                              value={presetPlaces.includes(editPlace) ? '' : editPlace}
                              onChange={(e) => setEditPlace(e.target.value)}
                            />
                          )}
                        </div>
                        <div className="field">
                          <label htmlFor="oet">Time window</label>
                          <select
                            className="input"
                            id="oet"
                            value={selTimeOptions.includes(editTime) ? editTime : selTimeOptions[0] || editTime}
                            onChange={(e) => setEditTime(e.target.value)}
                          >
                            {selTimeOptions.map((tm) => (
                              <option key={tm} value={tm}>
                                {tm}
                              </option>
                            ))}
                          </select>
                        </div>
                      </div>
                      <div className="row" style={{ justifyContent: 'flex-end' }}>
                        <button
                          type="button"
                          className="btn ghost sm"
                          onClick={() => setOpenBox('')}
                        >
                          Cancel
                        </button>
                        <button type="submit" className="btn sm">
                          Save new time
                        </button>
                      </div>
                    </form>
                  )}

                  {openBox === 'cancel' && (
                    <div
                      className="card flat stack"
                      style={{ background: 'var(--red-bg)' }}
                    >
                      <b className="small">Cancel request {sel.id}?</b>
                      <p className="tiny">
                        The seller will be notified and the reserved stock will be returned.
                      </p>
                      <div className="row" style={{ justifyContent: 'flex-end' }}>
                        <button
                          type="button"
                          className="btn ghost sm"
                          onClick={() => setOpenBox('')}
                        >
                          Keep request
                        </button>
                        <button
                          type="button"
                          className="btn danger sm"
                          onClick={() => {
                            cancelOrderRequest(sel.id)
                            setOpenBox('')
                          }}
                        >
                          Yes, cancel it
                        </button>
                      </div>
                    </div>
                  )}
                </>
              )}

              {!selIsOfficial && sel.tab !== 'Active' && (
                <div className="row wrap" style={{ justifyContent: 'flex-end' }}>
                  <button
                    type="button"
                    className="btn ghost sm"
                    onClick={() =>
                      openModal({
                        kind: 'order-report',
                        orderId: sel.id,
                        role: 'buyer',
                        targetEmail: sel.sellerEmail,
                        targetName: sel.seller,
                        itemTitle: sel.title,
                      })
                    }
                  >
                    <Icon name="flag" size="sm" />
                    Report issue with seller
                  </button>
                </div>
              )}

              {/* Order review section for completed pickups (orderReviewHtml in js/9e-reviews-ranking.js) */}
              {orderRateable(sel) && (
                <OrderReviewSection
                  order={sel}
                  reviewOf={reviewOf}
                  submitReview={submitReview}
                  byId={byId}
                />
              )}
            </div>
          )}
        </div>
      )}
    </>
  )
}

function OrderReviewSection({
  order,
  reviewOf,
  submitReview,
  byId,
}: {
  order: OrderRecord
  reviewOf: ReturnType<typeof useTindahan>['reviewOf']
  submitReview: ReturnType<typeof useTindahan>['submitReview']
  byId: ReturnType<typeof useTindahan>['byId']
}) {
  const [editingKey, setEditingKey] = useState<string | null>(null)
  const [stars, setStars] = useState<number>(0)
  const [text, setText] = useState<string>('')

  const seenPids = new Set<string>()
  const lines = (order.lines || []).filter(
    (l) => l.pid && byId(l.pid) && !seenPids.has(l.pid) && seenPids.add(l.pid),
  )

  const renderBox = (
    kind: 'seller' | 'item',
    pid: string,
    title: string,
    sub: string,
  ) => {
    const k = kind === 'seller' ? `seller:${order.id}` : `item:${order.id}:${pid}`
    const ex = reviewOf(kind, order.id, pid)
    const isEditing = editingKey === k

    if (ex && !isEditing) {
      return (
        <div
          key={k}
          className="card flat stack"
          style={{ gap: 6, background: 'var(--bg)' }}
        >
          <div className="row between wrap">
            <div>
              <b className="small">{title}</b>
              <div className="tiny muted">{sub}</div>
            </div>
            <div className="row" style={{ gap: 8 }}>
              <StarsDisplay avg={ex.stars} />
              <b className="small tnum">{ex.stars}.0</b>
              <button
                type="button"
                className="btn ghost sm"
                onClick={() => {
                  setEditingKey(k)
                  setStars(ex.stars)
                  setText(ex.text || '')
                }}
              >
                Edit
              </button>
            </div>
          </div>
          {ex.text ? (
            <p className="small" style={{ overflowWrap: 'anywhere' }}>
              {ex.text}
            </p>
          ) : (
            <p className="tiny muted">No written review.</p>
          )}
        </div>
      )
    }

    return (
      <form
        key={k}
        className="card flat stack"
        style={{ gap: 8, background: 'var(--bg)' }}
        onSubmit={(e) => {
          e.preventDefault()
          const curStars = isEditing ? stars : stars
          if (submitReview(kind, order.id, pid, curStars, text)) {
            setEditingKey(null)
            setStars(0)
            setText('')
          }
        }}
      >
        <div>
          <b className="small">{title}</b>
          <div className="tiny muted">{sub}</div>
        </div>
        <StarPicker
          value={editingKey === k ? stars : 0}
          label={title}
          onChange={(val) => {
            if (editingKey !== k) {
              setEditingKey(k)
              setText(ex?.text || '')
            }
            setStars(val)
          }}
        />
        <label className="field">
          <span className="tiny muted">Short review (optional, up to 500 characters)</span>
          <textarea
            className="input"
            maxLength={500}
            rows={2}
            placeholder="Was the item as described? How was the campus meetup?"
            value={editingKey === k ? text : ''}
            onChange={(e) => {
              if (editingKey !== k) {
                setEditingKey(k)
              }
              setText(e.target.value)
            }}
          />
        </label>
        <div className="row" style={{ justifyContent: 'flex-end', gap: 8 }}>
          {ex && (
            <button
              type="button"
              className="btn ghost sm"
              onClick={() => {
                setEditingKey(null)
                setStars(0)
                setText('')
              }}
            >
              Cancel
            </button>
          )}
          <button type="submit" className="btn sm">
            {ex ? 'Update rating' : 'Submit rating'}
          </button>
        </div>
      </form>
    )
  }

  return (
    <div
      className="stack"
      style={{ gap: 10, borderTop: '1px solid var(--line)', paddingTop: 12 }}
    >
      <div>
        <b>Rate this completed pickup</b>
        <p className="tiny muted">
          Your rating helps other RTU students know who and what to trust.
        </p>
      </div>
      {renderBox(
        'seller',
        '',
        `Rate seller: ${order.seller}`,
        'How was the communication and campus meetup?',
      )}
      {lines.map((l) => {
        const p = byId(l.pid)
        if (!p) return null
        return renderBox(
          'item',
          p.id,
          `Rate item: ${p.name}`,
          'Was it in the condition described?',
        )
      })}
    </div>
  )
}
