import { useEffect, useState } from 'react'
import { Icon } from '@/common/components/Icon'
import { Link, navigate } from '@/common/router/router'
import { useTheme } from '@/common/hooks/useTheme'
import { useAuth } from '@/features/authentication'

import { imgSrc } from '@/features/marketplace/imageStore'
import logoImg from '@/assets/logo-removebg-preview.png'
import './Topbar.css'

const MEMBER_LABEL: Record<User['member_type'], string> = {
  student: 'Student',
  faculty: 'Faculty',
  personnel: 'Personnel',
}

interface TopbarProps {
  user: User
  onMenuClick: () => void
  onCloseMobileMenu?: () => void
}

/**
 * TOP BAR for signed-in pages: menu button (phones), search, messages,
 * notifications, cart and the user menu.
 */
export function Topbar({ user, onMenuClick, onCloseMobileMenu }: TopbarProps) {
  const { logout } = useAuth()
  const { theme, toggle } = useTheme()
  const {
    currentAccount,
    isStaffUser,
    cartCount,
    unreadMessagesCount,
    browseState,
    browseFor,
    markAllNotifsRead,
    openNotification,
  } = useTindahan()

  const [open, setOpen] = useState<'' | 'bell' | 'user'>('')
  const [query, setQuery] = useState(browseState.q || '')

  useEffect(() => {
    setQuery(browseState.q || '')
  }, [browseState.q])

  useEffect(() => {
    if (!open) return
    const onDocClick = () => setOpen('')
    window.addEventListener('click', onDocClick)
    return () => window.removeEventListener('click', onDocClick)
  }, [open])

  const submitSearch = (e: React.FormEvent) => {
    e.preventDefault()
    const q = query.trim()
    browseFor({ q, goBrowse: true })
    onCloseMobileMenu?.()
  }

  const toggleMenu = (which: 'bell' | 'user') => setOpen((cur) => (cur === which ? '' : which))

  const notifs = currentAccount?.notifs || []
  const unreadNotifs = notifs.filter((n) => !n.read).length
  const displayName = currentAccount?.name || user.display_name
  const avatarSrc =
    (currentAccount?.avatar ? imgSrc(currentAccount.avatar) : null) || user.profile_photo || null
  const init = initialsOf(displayName)
  const subLabel = currentAccount?.aff || MEMBER_LABEL[user.member_type] || 'Student'

  return (
    <header className="top">
      <div className="top-start">
        <Link
          to="marketplace"
          className="logo-link"
          aria-label="RTU Tindahan Home"
          onClick={() => {
            setOpen('')
            onCloseMobileMenu?.()
          }}
        >
          <img src={logoImg} alt="RTU Tindahan" className="topbar-logo" />
        </Link>
        <button
          type="button"
          className="menu-btn"
          aria-label="Open menu"
          onClick={() => {
            setOpen('')
            onMenuClick()
          }}
        >
          <Icon name="menu" />
        </button>
      </div>

      <form className="search" role="search" onSubmit={submitSearch}>
        <Icon name="search" size="sm" />
        <input
          placeholder="Search textbooks, uniforms, calculators or course codes..."
          aria-label="Search marketplace"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onFocus={() => onCloseMobileMenu?.()}
        />
      </form>

      <div className="top-end">
        <div className="right-actions">
          <Link
            to="messages"
            className="iconbtn"
            aria-label="Messages"
            title="Messages"
            onClick={() => {
              setOpen('')
              onCloseMobileMenu?.()
            }}
          >
            <Icon name="msg" />
            {unreadMessagesCount > 0 && <span className="dot">{unreadMessagesCount}</span>}
          </Link>

          <div className="rel" onClick={(e) => e.stopPropagation()}>
            <button
              type="button"
              className={`iconbtn bellbtn ${open === 'bell' ? 'is-open' : ''}`.trim()}
              aria-label="Notifications"
              aria-haspopup="true"
              aria-expanded={open === 'bell'}
              onClick={() => {
                toggleMenu('bell')
                onCloseMobileMenu?.()
              }}
            >
              <Icon name="bell" />
              {unreadNotifs > 0 && <span className="dot">{unreadNotifs}</span>}
            </button>
            {open === 'bell' && (
              <div className="menu notif-menu" role="region" aria-label="Notifications">
                <div className="row between" style={{ padding: '6px 10px' }}>
                  <b className="small">Notifications</b>
                  {unreadNotifs > 0 && (
                    <button
                      type="button"
                      className="link"
                      onClick={() => markAllNotifsRead()}
                    >
                      Mark all read
                    </button>
                  )}
                </div>
                <hr />
                {notifs.length ? (
                  notifs.slice(0, 12).map((n, i) => (
                    <button
                      key={i}
                      type="button"
                      className={`mi notif ${n.read ? 'read' : 'unread'}`}
                      onClick={() => {
                        setOpen('')
                        openNotification(i)
                      }}
                    >
                      <span className="nbul" aria-hidden="true" />
                      <span className="grow">
                        <span className="ntxt">{n.t}</span>
                        <span className="tiny muted">{n.at ? ago(n.at) : n.ts}</span>
                      </span>
                    </button>
                  ))
                ) : (
                  <div className="small muted" style={{ padding: 12 }}>
                    Nothing yet. Updates about your requests show up here.
                  </div>
                )}
              </div>
            )}
          </div>

          <Link
            to="cart"
            className="iconbtn"
            aria-label="Pickup cart"
            title="Pickup cart"
            onClick={() => {
              setOpen('')
              onCloseMobileMenu?.()
            }}
          >
            <Icon name="cart" />
            {cartCount > 0 && <span className="dot">{cartCount}</span>}
          </Link>

          <div className="rel" onClick={(e) => e.stopPropagation()}>
            <button
              type="button"
              className="userbtn"
              aria-haspopup="true"
              aria-expanded={open === 'user'}
              onClick={() => {
                toggleMenu('user')
                onCloseMobileMenu?.()
              }}
            >
              <span className="avw av-sm">
                <span className={`avatar ${isStaffUser ? 'staff' : ''}`.trim()}>
                  {avatarSrc ? <img className="avimg" src={avatarSrc} alt={displayName} /> : init}
                </span>
              </span>
              <span className="user-text-slot">
                <b title={displayName} className="user-name">
                  {displayName}
                </b>
                <span title={`${subLabel} · Pasig`} className="user-sub">
                  {subLabel} · Pasig
                </span>
              </span>
            </button>
            {open === 'user' && (
              <div className="menu" role="menu">
                <div style={{ padding: '8px 10px' }}>
                  <b style={{ fontSize: 13 }}>{displayName}</b>
                  <div className="tiny muted" style={{ overflowWrap: 'anywhere' }}>
                    {currentAccount?.email || user.email} · {currentAccount?.sid || user.id}
                  </div>
                </div>
                <hr />
                <Link to="profile" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="user" /> My profile
                </Link>
                <Link to="orders" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="order" /> My orders
                </Link>
                <Link to="shop" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="shop" /> {isStaffUser ? 'Admin | RTU Shop' : 'My shop'}
                </Link>
                <Link to="favorites" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="heart" /> Favorites ({currentAccount?.favs.length || 0})
                </Link>
                <Link to="messages" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="msg" /> Messages
                  {unreadMessagesCount > 0 ? ` (${unreadMessagesCount})` : ''}
                </Link>
                <Link to="help" className="mi" role="menuitem" onClick={() => setOpen('')}>
                  <Icon name="info" /> Help Center & Manual
                </Link>
                <button type="button" className="mi" role="menuitem" onClick={toggle}>
                  <Icon name={theme === 'dark' ? 'sun' : 'moon'} />{' '}
                  {theme === 'dark' ? 'Switch to light theme' : 'Switch to dark theme'}
                </button>
                {isStaffUser && (
                  <>
                    <hr />
                    <Link to="moderation" className="mi" role="menuitem" onClick={() => setOpen('')}>
                      <Icon name="flag" /> Listing approval
                    </Link>
                    <Link to="staff" className="mi" role="menuitem" onClick={() => setOpen('')}>
                      <Icon name="sliders" /> Staff tools
                    </Link>
                  </>
                )}
                <hr />
                <button
                  type="button"
                  className="mi"
                  role="menuitem"
                  style={{ color: 'var(--red)' }}
                  onClick={() => {
                    setOpen('')
                    void logout().then(() => navigate(''))
                  }}
                >
                  <Icon name="logout" /> Sign out
                </button>
              </div>
            )}
          </div>
        </div>
      </div>
    </header>
  )
}

import { useState, useEffect, type ReactNode } from 'react'
import type { User } from '@/common/types'
import { GlobalOverlays } from '@/features/marketplace/components/GlobalOverlays'
import { Sidebar } from '../components/Sidebar'
import { Topbar } from '../components/Topbar'
import { SiteFooter } from '../components/SiteFooter'
import './AppLayout.css'

interface AppLayoutProps {
  user: User
  activePath: string
  children: ReactNode
}

/**
 * LAYOUT for signed-in pages:
 *   [ Topbar (full width, fixed)         ]
 *   [ Sidebar (fixed, collapsible) ] [ page content ]
 *               [ footer under page only ]
 * Page components only provide the content; this layout adds everything else.
 *
 * The exact height of the fixed topbar + scalloped awning is measured via
 * ResizeObserver and exposed as --header-offset so the sidebar and <main>
   can position themselves precisely below it without any overlap or guesswork.
 */
export function AppLayout({ user, activePath, children }: AppLayoutProps) {
  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false)
  const [isSidebarOpen, setIsSidebarOpen] = useState(
    typeof window !== 'undefined' ? window.innerWidth >= 768 : true
  )

  // Re-evaluate default open/closed state when the viewport crosses 768px.
  // Close the mobile drawer when resizing to desktop (>= 768px).
  useEffect(() => {
    const onResize = () => {
      const wasMobile = window.innerWidth < 768
      setIsSidebarOpen(window.innerWidth >= 768)
      if (!wasMobile && window.innerWidth >= 768 && isMobileMenuOpen) {
        setIsMobileMenuOpen(false)
      }
    }
    window.addEventListener('resize', onResize)
    return () => window.removeEventListener('resize', onResize)
  }, [isMobileMenuOpen])

  // Measure the fixed topbar+awning height (--header-offset) AND navbar-only height (--navbar-height).
  // --header-offset stays untouched for desktop and <main>. --navbar-height is mobile-only.
  useEffect(() => {
    if (typeof window === 'undefined' || !ResizeObserver) return

    const setOffset = () => {
      const el = document.querySelector('.top')
      if (!el) return

      const rect = el.getBoundingClientRect()
      // --navbar-height: just the navy bar content height (mobile drawer anchors here)
      document.documentElement.style.setProperty(
        '--navbar-height',
        `${Math.ceil(rect.height)}px`
      )
      // --header-offset: navbar + scalloped awning strip (+10px). Untouched for desktop/<main>.
      document.documentElement.style.setProperty(
        '--header-offset',
        `${Math.ceil(rect.height) + 10}px`
      )
    }

    setOffset()
    const ro = new ResizeObserver(() => setOffset())
    ro.observe(document.documentElement)
    window.addEventListener('resize', setOffset)
    return () => {
      ro.disconnect()
      window.removeEventListener('resize', setOffset)
    }
  }, [])

  // Mobile drawer (< 900px): lock background scroll and close on Escape
  useEffect(() => {
    const isMobile = window.innerWidth < 900
    if (!isMobileMenuOpen || !isMobile) return

    // Save scroll position so the page doesn't jump when we lock it
    const scrollY = window.scrollY

    document.body.style.overflow = 'hidden'
    document.body.style.position = 'fixed'
    document.body.style.top = `-${scrollY}px`
    document.body.style.left = '0'
    document.body.style.width = '100%'
    document.documentElement.style.overflow = 'hidden'
    document.documentElement.style.touchAction = 'none'
    document.body.style.touchAction = 'none'

    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setIsMobileMenuOpen(false)
    }

    window.addEventListener('keydown', onKeyDown)

    return () => {
      document.body.style.removeProperty('overflow')
      document.body.style.removeProperty('position')
      document.body.style.removeProperty('top')
      document.body.style.removeProperty('left')
      document.body.style.removeProperty('width')
      document.documentElement.style.removeProperty('overflow')
      document.documentElement.style.removeProperty('touchAction')
      document.body.style.removeProperty('touchAction')
      window.scrollTo(0, scrollY)
      window.removeEventListener('keydown', onKeyDown)
    }
  }, [isMobileMenuOpen])

  // Safety net: close the mobile drawer whenever the route changes,
  // so topbar navigation always dismisses the drawer.
  useEffect(() => {
    setIsMobileMenuOpen(false)
  }, [activePath])
  return (
    <div className={`app ${isSidebarOpen ? 'sidebar-open' : ''} ${isMobileMenuOpen ? 'mobile-drawer-open' : ''}`}>
      <Topbar
        user={user}
        onMenuClick={() => setIsMobileMenuOpen((prev) => !prev)}
        onCloseMobileMenu={() => setIsMobileMenuOpen(false)}
      />
      {/* Mobile-only click-outside backdrop — renders only when drawer is open (< 768px) */}
      {isMobileMenuOpen && (
        <div
          className="mobile-backdrop"
          onClick={() => setIsMobileMenuOpen(false)}
          aria-hidden="true"
        />
      )}
      <div className="main">
        <Sidebar
          activePath={activePath}
          role={user.role}
          open={isMobileMenuOpen}
          isOpen={isSidebarOpen}
          onClose={() => setIsMobileMenuOpen(false)}
          onToggle={() => setIsSidebarOpen((prev) => !prev)}
        />
        <div className="page-wrap">
          <main className="page">
            <div className="page-content-shell">{children}</div>
          </main>
          <SiteFooter variant="member" />
        </div>
      </div>
      <GlobalOverlays />
    </div>
  )
}
