const FR_STATUS_LABELS: Record<string, string> = {
  open: 'Open',
  reviewed: 'Reviewed',
  dismissed: 'Dismissed',
  actioned: 'Action taken',
}

function postTitle(f: ForumPost): string {
  const t = String(f.title || '').trim()
  if (t) return t
  const b = String(f.body || '').trim().replace(/\s+/g, ' ')
  return b ? (b.length > 70 ? b.slice(0, 70).trim() + '...' : b) : 'Untitled post'
}

export function ForumReportsPage() {
  const {
    db,
    isStaffUser,
    openPost,
    setForumReportStatus,
    openModal,
  } = useTindahan()

  const [frTab, setFrTab] = useState<'open' | 'reviewed' | 'dismissed' | 'actioned' | 'all'>('open')

  if (!isStaffUser) {
    return (
      <div className="card stack">
        <b>This page is for RTU staff accounts only.</b>
      </div>
    )
  }

  const tabs: ['open' | 'reviewed' | 'dismissed' | 'actioned' | 'all', string][] = [
    ['open', 'Open'],
    ['reviewed', 'Reviewed'],
    ['dismissed', 'Dismissed'],
    ['actioned', 'Action taken'],
    ['all', 'All'],
  ]

  const cnt = (k: string) =>
    db.freports.filter((r) => k === 'all' || r.status === k).length
  const list = db.freports.filter((r) => frTab === 'all' || r.status === frTab)

  return (
    <div className="fcol wide">
      <div>
        <span className="eyebrow">STAFF · MEMBER & FORUM REPORTS</span>
        <h1>Member, Order & Forum Reports</h1>
        <p className="muted small">
          Reports sent by buyers and sellers about meetup/order problems (such as no-shows) as well
          as reported forum posts and comments.
        </p>
      </div>

      <div className="tabs" role="tablist" aria-label="Report status">
        {tabs.map((t) => (
          <button
            key={t[0]}
            type="button"
            className={`tab ${frTab === t[0] ? 'on' : ''}`.trim()}
            role="tab"
            aria-selected={frTab === t[0]}
            onClick={() => setFrTab(t[0])}
          >
            {t[1]} <span className="tnum">({cnt(t[0])})</span>
          </button>
        ))}
      </div>

      {list.length ? (
        list.map((r) => {
          const f = db.feed.find((x) => x.id === r.postId)
          const c = f && r.commentId ? f.comments.find((x) => x.id === r.commentId) : undefined
          const gone = !f || (Boolean(r.commentId) && (!c || c.deleted))
          const live = gone ? null : c || f
          const text = gone ? r.snap.text : c ? c.text : f?.body || r.snap.text

          return (
            <article
              key={r.id}
              className="card stack frcard"
              id={`rep-${r.id}`}
              style={{ gap: 10 }}
            >
              <div className="row between wrap" style={{ gap: 8 }}>
                <div className="row wrap" style={{ gap: 6 }}>
                  <span
                    className={`pill ${
                      r.status === 'open'
                        ? 'red'
                        : r.status === 'actioned'
                          ? 'navy'
                          : ''
                    }`.trim()}
                  >
                    {FR_STATUS_LABELS[r.status] || r.status}
                  </span>
                  <span className={`pill ${r.type === 'order' ? 'yellow' : ''}`.trim()}>
                    {r.type === 'order'
                      ? `Order Report (${r.reporterRole === 'seller' ? 'Seller → Buyer' : 'Buyer → Seller'})`
                      : r.type === 'post'
                        ? 'Post'
                        : 'Comment'}
                  </span>
                  <b>{r.reason}</b>
                </div>
                <span className="tiny muted">{stamp(r.ts)}</span>
              </div>

              {r.details && (
                <p className="small" style={{ overflowWrap: 'anywhere' }}>
                  {r.details}
                </p>
              )}

              <div className="frquote">
                <div className="tiny muted">
                  {r.type === 'order'
                    ? `Transaction / Meetup Report · ${r.snap.title}`
                    : `Reported ${r.type}${r.type === 'comment' ? ` on the post "${r.snap.title}"` : ''}${
                        gone
                          ? ' (no longer exists, copy saved when reported)'
                          : live && live.hidden
                            ? ' (currently hidden)'
                            : ''
                      }`}
                </div>
                {(r.type === 'post' || r.type === 'order') && (
                  <b>{r.type === 'order' ? r.snap.title : f ? postTitle(f) : r.snap.title}</b>
                )}
                <div
                  className="small"
                  style={{ overflowWrap: 'anywhere', whiteSpace: 'pre-wrap' }}
                >
                  {String(text || '').slice(0, 400) || <i>No text</i>}
                </div>
                {r.snap.imgs > 0 && (
                  <div className="tiny muted">
                    {r.snap.imgs} photo{r.snap.imgs === 1 ? '' : 's'} attached
                  </div>
                )}
              </div>

              <div className="frmeta">
                <div>
                  <span className="tiny muted">REPORTED BY</span>
                  <br />
                  {r.reporterName}
                  <br />
                  <span className="tiny muted">{r.reporterEmail}</span>
                </div>
                <div>
                  <span className="tiny muted">REPORTED USER</span>
                  <br />
                  {r.targetName}
                  {r.targetEmail && (
                    <>
                      <br />
                      <span className="tiny muted">{r.targetEmail}</span>
                    </>
                  )}
                </div>
                <div>
                  <span className="tiny muted">SUBMITTED</span>
                  <br />
                  {stamp(r.ts)}
                </div>
                <div>
                  <span className="tiny muted">STATUS</span>
                  <br />
                  {FR_STATUS_LABELS[r.status] || r.status}
                  {r.handledBy && (
                    <>
                      <br />
                      <span className="tiny muted">
                        by {r.handledBy}
                        {r.handledAt ? `, ${stamp(r.handledAt)}` : ''}
                      </span>
                    </>
                  )}
                </div>
              </div>

              {r.action ? (
                <div className="small">
                  <b>Action:</b> {FR_ACTIONS[r.action] ? FR_ACTIONS[r.action][0] : r.action}
                  {r.note ? ` · ${r.note}` : ''}
                </div>
              ) : (
                r.note && (
                  <div className="small">
                    <b>Note:</b> {r.note}
                  </div>
                )
              )}

              <div className="row wrap" style={{ gap: 6 }}>
                {r.type === 'order' ? (
                  r.targetEmail ? (
                    <button
                      type="button"
                      className="btn ghost sm"
                      onClick={() => openModal({ kind: 'admin-user', email: r.targetEmail! })}
                    >
                      Manage reported account ({r.targetName})
                    </button>
                  ) : null
                ) : (
                  <button
                    type="button"
                    className="btn ghost sm"
                    disabled={gone}
                    onClick={() => openPost(r.postId, r.commentId)}
                  >
                    Open content
                  </button>
                )}
                {r.status === 'open' ? (
                  <>
                    <button
                      type="button"
                      className="btn ghost sm"
                      onClick={() => setForumReportStatus(r.id, 'reviewed')}
                    >
                      Mark reviewed
                    </button>
                    <button
                      type="button"
                      className="btn ghost sm"
                      onClick={() => setForumReportStatus(r.id, 'dismissed')}
                    >
                      Dismiss report
                    </button>
                  </>
                ) : (
                  <button
                    type="button"
                    className="btn ghost sm"
                    onClick={() => setForumReportStatus(r.id, 'open')}
                  >
                    Reopen
                  </button>
                )}
                {r.type !== 'order' && (
                  <button
                    type="button"
                    className="btn sm"
                    onClick={() =>
                      openModal({ kind: 'forum-mod-action', reportId: r.id })
                    }
                  >
                    Take moderation action
                  </button>
                )}
              </div>
            </article>
          )
        })
      ) : (
        <div
          className="card stack"
          style={{ alignItems: 'center', textAlign: 'center', padding: 36 }}
        >
          <span style={{ color: 'var(--blue)' }}>
            <Icon name="check" />
          </span>
          <b>{frTab === 'open' ? 'No open reports' : 'No reports here'}</b>
          <p className="small muted">Reports from members will appear in this list.</p>
        </div>
      )}

      <section className="card stack" style={{ gap: 8 }}>
        <b>
          Moderation log <span className="tiny muted tnum">({db.modlog.length})</span>
        </b>
        {db.modlog.length ? (
          <div style={{ overflowX: 'auto' }}>
            <table className="tbl">
              <thead>
                <tr>
                  <th>When</th>
                  <th>Staff</th>
                  <th>Action</th>
                  <th>Target</th>
                  <th>Outcome</th>
                </tr>
              </thead>
              <tbody>
                {db.modlog.slice(0, 25).map((m) => (
                  <tr key={m.id}>
                    <td className="tnum">{stamp(m.ts)}</td>
                    <td>{m.by}</td>
                    <td>{m.kind}</td>
                    <td>
                      {m.type} by {m.targetName || 'unknown'}
                    </td>
                    <td>
                      {m.outcome || ''}
                      {m.note ? ` · ${m.note}` : ''}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        ) : (
          <p className="small muted">No moderation actions yet.</p>
        )}
      </section>
    </div>
  )
}

import { useEffect, useState } from 'react'
import { Icon } from '@/common/components/Icon'

import { REACTS, TOPICS, ago, peso, priceLabel, stamp } from '../constants'

function postTitle(f: ForumPost): string {
  const t = String(f.title || '').trim()
  if (t) return t
  const b = String(f.body || '').trim().replace(/\s+/g, ' ')
  return b ? (b.length > 70 ? b.slice(0, 70).trim() + '...' : b) : 'Untitled post'
}

export function ForumPage() {
  const {
    db,
    currentAccount,
    isStaffUser,
    allOnSale,
    acctByEmail,
    openProduct,
    openPost,
    createForumPost,
    deleteForumPost,
    toggleForumReaction,
    togglePinPost,
    toggleHidePost,
    openModal,
  } = useTindahan()

  const [forumTab, setForumTab] = useState<'all' | 'post' | 'listing' | 'notice'>('all')
  const [forumSort, setForumSort] = useState<'latest' | 'popular'>('latest')
  const [fpLimit, setFpLimit] = useState(10)

  const [title, setTitle] = useState('')
  const [body, setBody] = useState('')
  const [topic, setTopic] = useState<string>(TOPICS[0])
  const [asNotice, setAsNotice] = useState(false)
  const [imgs, setImgs] = useState<string[]>([])
  const [openMenuKey, setOpenMenuKey] = useState<string | null>(null)

  useEffect(() => {
    if (!openMenuKey) return
    const onDocClick = () => setOpenMenuKey(null)
    window.addEventListener('click', onDocClick)
    return () => window.removeEventListener('click', onDocClick)
  }, [openMenuKey])

  if (!currentAccount) return null

  const fpVisible = (f: ForumPost) =>
    !f.hidden || isStaffUser || f.authorEmail === currentAccount.email
  const cmLive = (c: { deleted?: boolean; hidden?: unknown }) => !c.deleted && !c.hidden
  const cmCount = (f: ForumPost) => (f.comments || []).filter(cmLive).length
  const popScore = (f: ForumPost) => Object.keys(f.reacts || {}).length + cmCount(f)

  const inTab = (f: ForumPost, k: 'all' | 'post' | 'listing' | 'notice') =>
    k === 'all' ||
    (k === 'post'
      ? f.kind === 'post'
      : k === 'listing'
        ? f.kind === 'student' || f.kind === 'school'
        : f.kind === 'notice')

  const vis = db.feed.filter(fpVisible)
  const list = vis
    .filter((f) => inTab(f, forumTab))
    .sort(
      (a, b) =>
        (b.pinned ? 1 : 0) - (a.pinned ? 1 : 0) ||
        (forumSort === 'popular' ? popScore(b) - popScore(a) : 0) ||
        String(b.ts).localeCompare(String(a.ts)),
    )
  const shown = list.slice(0, fpLimit)
  const nOpenReports = isStaffUser
    ? db.freports.filter((r) => r.status === 'open').length
    : 0
  const notice = isStaffUser && asNotice

  const tabs: ['all' | 'post' | 'listing' | 'notice', string][] = [
    ['all', 'All posts'],
    ['post', 'Discussions'],
    ['listing', 'New listings'],
    ['notice', 'Notices'],
  ]

  return (
    <div className="fcol">
      <div>
        <span className="eyebrow">COMMUNITY · FORUM</span>
        <h1>Campus forum</h1>
        <p className="muted small">
          Ask questions, share tips and talk with other RTU Pasig students. Open a post to read it
          all and join the discussion.
        </p>
      </div>

      {nOpenReports > 0 && (
        <div
          className="note"
          role="status"
          style={{ background: 'var(--red-bg)', borderColor: 'transparent' }}
        >
          <Icon name="flag" />
          <span>
            <b>
              {nOpenReports} forum report{nOpenReports === 1 ? ' is' : 's are'} waiting.
            </b>{' '}
            <button
              type="button"
              className="link"
              onClick={() => navigate('forum-reports')}
            >
              Open Forum Reports
            </button>
          </span>
        </div>
      )}

      <form
        className="card stack"
        style={{ gap: 10 }}
        noValidate
        onSubmit={(e) => {
          e.preventDefault()
          if (createForumPost(title, body, topic, notice, imgs)) {
            setTitle('')
            setBody('')
            setImgs([])
            setAsNotice(false)
            setForumSort('latest')
          }
        }}
      >
        <div className="row" style={{ alignItems: 'flex-start' }}>
          <AvatarBadge account={currentAccount} isStaff={isStaffUser} size="md" />
          <div className="stack grow" style={{ gap: 8, minWidth: 0 }}>
            <input
              className="input"
              id="fp-title"
              maxLength={120}
              value={title}
              placeholder={notice ? 'Notice title' : 'Title (optional)'}
              aria-label={notice ? 'Notice title' : 'Post title'}
              autoComplete="off"
              onChange={(e) => setTitle(e.target.value)}
            />
            <textarea
              className="input"
              id="fp-text"
              maxLength={2000}
              placeholder={`What is on your mind, ${currentAccount.first}?`}
              aria-label="Write a post"
              value={body}
              onChange={(e) => setBody(e.target.value)}
            />
          </div>
        </div>

        <PhotoUploader
          ids={imgs}
          max={IMG_RULES.post}
          label="Photos (optional)"
          onChange={setImgs}
        />

        <div className="row between wrap">
          <span className="row wrap" style={{ gap: 10 }}>
            <label className="row small" style={{ gap: 6 }}>
              Topic{' '}
              <select
                className="input"
                id="fp-topic"
                style={{ width: 'auto' }}
                disabled={notice}
                value={topic}
                onChange={(e) => setTopic(e.target.value)}
              >
                {TOPICS.map((t) => (
                  <option key={t} value={t}>
                    {t}
                  </option>
                ))}
              </select>
            </label>
            {isStaffUser && (
              <label className="check">
                <input
                  type="checkbox"
                  id="fp-notice"
                  checked={asNotice}
                  onChange={(e) => setAsNotice(e.target.checked)}
                />
                Official notice (alerts every member)
              </label>
            )}
          </span>
          <button type="submit" className="btn yellow">
            <Icon name="send" size="sm" />
            {notice ? 'Post notice' : 'Post'}
          </button>
        </div>
      </form>

      <div className="row between wrap" style={{ gap: 10 }}>
        <div className="tabs" role="tablist" aria-label="Forum filter">
          {tabs.map((t) => (
            <button
              key={t[0]}
              type="button"
              className={`tab ${forumTab === t[0] ? 'on' : ''}`.trim()}
              role="tab"
              aria-selected={forumTab === t[0]}
              onClick={() => setForumTab(t[0])}
            >
              {t[1]}
            </button>
          ))}
        </div>
        <div className="row" style={{ gap: 6 }} role="group" aria-label="Sort posts">
          <button
            type="button"
            className={`btn sm ${forumSort === 'latest' ? '' : 'ghost'}`.trim()}
            aria-pressed={forumSort === 'latest'}
            onClick={() => setForumSort('latest')}
          >
            Latest
          </button>
          <button
            type="button"
            className={`btn sm ${forumSort === 'popular' ? '' : 'ghost'}`.trim()}
            aria-pressed={forumSort === 'popular'}
            onClick={() => setForumSort('popular')}
          >
            Popular
          </button>
        </div>
      </div>

      <p className="tiny muted" role="status">
        {forumSort === 'popular'
          ? 'Popular puts posts with the most reactions plus comments first. Pinned posts stay on top.'
          : 'Latest shows the newest posts first. Pinned posts stay on top.'}
      </p>

      {shown.length ? (
        shown.map((f) => {
          const author = f.authorEmail ? acctByEmail(f.authorEmail) : undefined
          const isStaffPost =
            f.kind === 'notice' || f.kind === 'school' || author?.role === 'moderator'
          const by = f.kind === 'notice' ? 'Office of Student Affairs' : f.by || 'RTU'
          const nrep = isStaffUser
            ? db.freports.filter((r) => r.status === 'open' && r.postId === f.id).length
            : 0
          const mine = f.authorEmail === currentAccount.email
          const reported = db.freports.some(
            (r) =>
              r.type === 'post' &&
              r.postId === f.id &&
              r.reporterEmail === currentAccount.email,
          )
          const nReact = Object.keys(f.reacts || {}).length
          const nCm = cmCount(f)
          const showTitle = f.kind !== 'student' && f.kind !== 'school'
          const bodyStr = String(f.body || '')
          const menuKey = `p:${f.id}`
          const menuOpen = openMenuKey === menuKey

          const liveProd =
            f.pid && allOnSale.some((p) => p.id === f.pid)
              ? allOnSale.find((p) => p.id === f.pid)
              : undefined
          const prodImgs = liveProd ? imgsOf(liveProd) : []
          const postImgs = f.imgs || []

          return (
            <article
              key={f.id}
              className={`card fpost ${f.pinned ? 'pinned' : ''} ${
                f.hidden ? 'is-hidden' : ''
              }`.trim()}
              id={`post-${f.id}`}
            >
              <div className="fhead">
                <AvatarBadge
                  account={f.kind !== 'notice' ? author : undefined}
                  name={by}
                  isStaff={isStaffPost}
                  size="md"
                />
                <div className="grow">
                  <div className="row wrap" style={{ gap: 6 }}>
                    <b style={{ fontSize: 14 }}>{by}</b>
                    {isStaffPost && <span className="pill navy">Staff</span>}
                    {f.kind === 'notice' && <span className="pill yellow">Notice</span>}
                    {(f.kind === 'student' || f.kind === 'school') && (
                      <span className={`srcpill ${f.kind}`}>
                        {f.kind === 'school' ? 'Official RTU Store' : 'Student seller'}
                      </span>
                    )}
                    {f.topic && f.kind === 'post' && <span className="pill">{f.topic}</span>}
                    {f.pinned && <span className="pill yellow">Pinned</span>}
                    {f.hidden && <span className="pill red">Hidden by staff</span>}
                    {nrep > 0 && (
                      <button
                        type="button"
                        className="pill red linkpill"
                        title="Open Forum Reports"
                        onClick={() => navigate('forum-reports')}
                      >
                        {nrep} open report{nrep === 1 ? '' : 's'}
                      </button>
                    )}
                  </div>
                  <div className="tiny muted" title={ago(f.ts)}>
                    {stamp(f.ts)} · {ago(f.ts)}
                    {f.edited && (
                      <>
                        {' '}
                        ·{' '}
                        <span className="edited-tag" title={`Edited ${stamp(f.edited)}`}>
                          edited
                        </span>
                      </>
                    )}
                  </div>
                </div>

                <div className="amenu-wrap" onClick={(e) => e.stopPropagation()}>
                  <button
                    type="button"
                    className="iconbtn amenu-btn"
                    aria-haspopup="menu"
                    aria-expanded={menuOpen}
                    aria-label="More actions for this post"
                    title="More actions"
                    onClick={() =>
                      setOpenMenuKey((cur) => (cur === menuKey ? null : menuKey))
                    }
                  >
                    <Icon name="dots" size="sm" />
                  </button>
                  {menuOpen && (
                    <div className="amenu" role="menu">
                      {mine && f.kind === 'post' && !f.hidden && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          onClick={() => {
                            setOpenMenuKey(null)
                            openModal({ kind: 'forum-edit-post', postId: f.id })
                          }}
                        >
                          <Icon name="edit" size="sm" />
                          <span>Edit post</span>
                        </button>
                      )}
                      {!mine && f.kind !== 'notice' && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          disabled={reported}
                          onClick={() => {
                            setOpenMenuKey(null)
                            openModal({
                              kind: 'forum-report',
                              postId: f.id,
                              commentId: null,
                            })
                          }}
                        >
                          <Icon name="flag" size="sm" />
                          <span>{reported ? 'Reported' : 'Report post'}</span>
                        </button>
                      )}
                      {isStaffUser && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          onClick={() => {
                            setOpenMenuKey(null)
                            togglePinPost(f.id)
                          }}
                        >
                          <Icon name="pin" size="sm" />
                          <span>{f.pinned ? 'Unpin post' : 'Pin post'}</span>
                        </button>
                      )}
                      {isStaffUser && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          onClick={() => {
                            setOpenMenuKey(null)
                            toggleHidePost(f.id)
                          }}
                        >
                          <Icon name="eye" size="sm" />
                          <span>{f.hidden ? 'Restore post' : 'Hide post'}</span>
                        </button>
                      )}
                      {(mine || isStaffUser) && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item danger"
                          onClick={() => {
                            setOpenMenuKey(null)
                            deleteForumPost(f.id)
                          }}
                        >
                          <Icon name="trash" size="sm" />
                          <span>Delete post</span>
                        </button>
                      )}
                    </div>
                  )}
                </div>
              </div>

              {showTitle && (
                <h3 className="ftitle">
                  <button
                    type="button"
                    className="ftitle-btn"
                    onClick={() => openPost(f.id)}
                  >
                    {postTitle(f)}
                  </button>
                </h3>
              )}

              {showTitle ? (
                bodyStr &&
                !(f.kind === 'post' && !String(f.title || '').trim() && bodyStr.length <= 70) && (
                  <div className="fbody">
                    {bodyStr.length > 280
                      ? bodyStr.slice(0, 280).trim() + '...'
                      : bodyStr}
                    {bodyStr.length > 280 && (
                      <>
                        {' '}
                        <button
                          type="button"
                          className="link"
                          onClick={() => openPost(f.id)}
                        >
                          Read more
                        </button>
                      </>
                    )}
                  </div>
                )
              ) : (
                <div className="fprod">
                  {prodImgs.length > 0 && imgSrc(prodImgs[0]) && (
                    <img
                      className="fprod-img"
                      src={imgSrc(prodImgs[0])}
                      alt={f.title || ''}
                    />
                  )}
                  <div className="grow" style={{ minWidth: 180 }}>
                    <b>
                      {f.svc ? 'New service: ' : 'New: '}
                      {f.title}
                    </b>
                    <div className="small muted" style={{ overflowWrap: 'anywhere' }}>
                      {bodyStr.length > 200 ? bodyStr.slice(0, 200) + '...' : bodyStr}
                    </div>
                    <div className="small tnum" style={{ marginTop: 2 }}>
                      {liveProd ? priceLabel(liveProd) : peso(f.price || 0)}
                      {f.tag ? ` · ${f.tag}` : ''}
                    </div>
                  </div>
                  {liveProd ? (
                    <button
                      type="button"
                      className="btn sm"
                      onClick={() => openProduct(liveProd.id)}
                    >
                      View {f.svc ? 'service' : 'listing'}
                    </button>
                  ) : (
                    <span className="pill">No longer available</span>
                  )}
                </div>
              )}

              {postImgs.length > 0 && (
                <div className={`fimgs n${Math.min(postImgs.length, 4)}`}>
                  {postImgs.map((id, idx) => {
                    const src = imgSrc(id)
                    if (!src) {
                      return (
                        <div key={`${id}-${idx}`} className="fimg-miss">
                          <Icon name="info" size="sm" />
                          <span>Image unavailable</span>
                        </div>
                      )
                    }
                    return (
                      <button
                        key={`${id}-${idx}`}
                        type="button"
                        className="fimg"
                        aria-label={`Open photo ${idx + 1} of ${postImgs.length}`}
                        onClick={() =>
                          openModal({ kind: 'forum-photo', postId: f.id, index: idx })
                        }
                      >
                        <img
                          src={src}
                          alt={`Photo ${idx + 1} attached to the post ${postTitle(f)}`}
                          loading="lazy"
                        />
                      </button>
                    )
                  })}
                </div>
              )}

              <div className="fcount">
                <span>
                  {nReact ? `${nReact} reaction${nReact === 1 ? '' : 's'}` : 'No reactions yet'}
                </span>
                <span>
                  {nCm ? `${nCm} comment${nCm === 1 ? '' : 's'}` : 'No comments yet'}
                </span>
              </div>

              <div className="freacts" role="group" aria-label="React to this post">
                {REACTS.map((r) => {
                  const myReact = f.reacts[currentAccount.email]
                  const n = Object.values(f.reacts || {}).filter((x) => x === r[0]).length
                  return (
                    <button
                      key={r[0]}
                      type="button"
                      className={`rbtn ${myReact === r[0] ? 'on' : ''}`.trim()}
                      aria-pressed={myReact === r[0]}
                      aria-label={`${r[2]}${n ? `, ${n}` : ''}`}
                      onClick={() => toggleForumReaction(f.id, r[0])}
                    >
                      <span className="e">{r[1]}</span>
                      {n ? <span className="tnum">{n}</span> : <span className="muted">{r[2]}</span>}
                    </button>
                  )
                })}
                <span className="grow" />
                <button
                  type="button"
                  className="btn ghost sm"
                  onClick={() => openPost(f.id)}
                >
                  <Icon name="msg" size="sm" />
                  {nCm ? `Comments (${nCm})` : 'Comment'}
                </button>
              </div>
            </article>
          )
        })
      ) : (
        <div
          className="card stack"
          style={{ alignItems: 'center', textAlign: 'center', padding: 40 }}
        >
          <span style={{ color: 'var(--blue)' }}>
            <Icon name="msg" />
          </span>
          <b>
            {forumTab === 'all' || forumTab === 'post'
              ? 'Be the first to post'
              : 'Nothing here yet'}
          </b>
          <p className="small muted" style={{ maxWidth: '48ch' }}>
            {forumTab === 'notice'
              ? 'Notices from RTU staff will show up here.'
              : forumTab === 'listing'
                ? 'When a listing is approved or the school adds an official product, it is announced here.'
                : 'Start a conversation above. Other students can react and reply.'}
          </p>
        </div>
      )}

      {list.length > shown.length && (
        <button
          type="button"
          className="btn ghost"
          style={{ alignSelf: 'center' }}
          onClick={() => setFpLimit((n) => n + 10)}
        >
          Show more posts
        </button>
      )}
    </div>
  )
}

import { useEffect, useState, type ReactNode } from 'react'

function postTitle(f: ForumPost): string {
  const t = String(f.title || '').trim()
  if (t) return t
  const b = String(f.body || '').trim().replace(/\s+/g, ' ')
  return b ? (b.length > 70 ? b.slice(0, 70).trim() + '...' : b) : 'Untitled post'
}

export function ForumPostPage() {
  const { params } = useRoute()
  const {
    db,
    currentAccount,
    isStaffUser,
    selectedPostId,
    highlightedCommentId,
    allOnSale,
    acctByEmail,
    openProduct,
    deleteForumPost,
    toggleForumReaction,
    togglePinPost,
    toggleHidePost,
    addForumComment,
    editForumComment,
    deleteForumComment,
    toggleHideComment,
    openModal,
  } = useTindahan()

  const [newComment, setNewComment] = useState('')
  const [replyToId, setReplyToId] = useState<string | null>(null)
  const [replyText, setReplyText] = useState('')
  const [editCmId, setEditCmId] = useState<string | null>(null)

  const [editCmErr, setEditCmErr] = useState('')
  const [openMenuKey, setOpenMenuKey] = useState<string | null>(null)

  useEffect(() => {
    if (!openMenuKey) return
    const onDocClick = () => setOpenMenuKey(null)
    window.addEventListener('click', onDocClick)
    return () => window.removeEventListener('click', onDocClick)
  }, [openMenuKey])

  if (!currentAccount) return null

  const postId = params.get('id') || selectedPostId
  const f = postId ? db.feed.find((x) => x.id === postId) : undefined
  const fpVisible =
    f && (!f.hidden || isStaffUser || f.authorEmail === currentAccount.email)

  if (!f || !fpVisible) {
    return (
      <div className="fcol">
        <button
          type="button"
          className="btn ghost sm"
          style={{ alignSelf: 'flex-start' }}
          onClick={() => navigate('forum')}
        >
          <Icon name="back" size="sm" />
          Back to the forum
        </button>
        <div
          className="card stack"
          style={{ alignItems: 'center', textAlign: 'center', padding: 40 }}
        >
          <span style={{ color: 'var(--blue)' }}>
            <Icon name="info" />
          </span>
          <b>This post is not available</b>
          <p className="small muted">It may have been deleted or hidden by staff.</p>

    )

  const author = f.authorEmail ? acctByEmail(f.authorEmail) : undefined
  const isStaffPost =
    f.kind === 'notice' || f.kind === 'school' || author?.role === 'moderator'
  const by = f.kind === 'notice' ? 'Office of Student Affairs' : f.by || 'RTU'
  const nrep = isStaffUser
    ? db.freports.filter((r) => r.status === 'open' && r.postId === f.id).length
    : 0
  const mine = f.authorEmail === currentAccount.email
  const reported = db.freports.some(
    (r) =>
      r.type === 'post' &&
      r.postId === f.id &&
      r.reporterEmail === currentAccount.email,
  )
  const cmLive = (c: ForumComment) => !c.deleted && !c.hidden

  const nCm = (f.comments || []).filter(cmLive).length
  const postMenuKey = `p:${f.id}`
  const postMenuOpen = openMenuKey === postMenuKey

  const liveProd =
    f.pid && allOnSale.some((p) => p.id === f.pid)

  const prodImgs = liveProd ? imgsOf(liveProd) : []
  const postImgs = f.imgs || []
  const bodyStr = String(f.body || '')

  const renderCommentTree = (parent: string | null, depth: number): ReactNode => {
    const children = (f.comments || []).filter((c) => (c.parent || null) === parent)
    if (!children.length) return null
    return children.map((c) => {
      const cMine = c.email === currentAccount.email
      const cAcct = acctByEmail(c.email)
      const cStaff = cAcct?.role === 'moderator'

        (r) =>
          r.type === 'comment' &&
          r.commentId === c.id &&
          r.reporterEmail === currentAccount.email,

      const cMenuKey = `c:${c.id}`
      const cMenuOpen = openMenuKey === cMenuKey
      const kids = renderCommentTree(c.id, depth + 1)

          <div className="bubc muted">
            <i>[deleted]</i>

            <i>[removed by moderators]</i>
          </div>
        )

            className="stack"
            style={{ gap: 6 }}
            onSubmit={(e) => {
              e.preventDefault()
              if (!editCmText.trim()) {
                setEditCmErr('A comment cannot be empty.')
                return
              }
              if (editForumComment(f.id, c.id, editCmText)) {
                setEditCmId(null)
                setEditCmText('')
                setEditCmErr('')
              }
            }}
          >
            <textarea
              className="input"
              maxLength={500}
              aria-label="Edit your comment"
              value={editCmText}
              onChange={(e) => {
                setEditCmText(e.target.value)
                setEditCmErr('')
              }}
            />
            {editCmErr && (
              <span className="err" role="alert">
                {editCmErr}
              </span>
            )}
            <div className="row" style={{ gap: 6 }}>
              <button type="submit" className="btn sm">
                Save
              </button>
              <button
                type="button"
                className="btn ghost sm"
                onClick={() => {
                  setEditCmId(null)
                  setEditCmErr('')
                }}
              >
                Cancel
              </button>
            </div>
          </form>
        )
      } else {
        bubble = (
          <div className="bubc">
            <b>
              {c.by}
              {cStaff && (

                  {' '}
                  <span className="pill navy">Staff</span>
                </>
              )}
              {c.hidden && (
                <>
                  {' '}
                  <span className="pill red">Hidden</span>
                </>
              )}
            </b>
            {c.text}
          </div>
        )
      }

      return (
        <div
          key={c.id}
          className={`fcm ${highlightedCommentId === c.id ? 'hl' : ''}`.trim()}
          id={`c-${c.id}`}
        >
          <AvatarBadge
            account={c.deleted ? undefined : cAcct}
            name={c.deleted ? '?' : c.by}
            isStaff={cStaff}
            size="sm"
          />
          <div style={{ minWidth: 0, flex: 1 }}>
            <div className="cm-row">
              <div className="cm-main">{bubble}</div>
              {!c.deleted && (
                <div className="amenu-wrap" onClick={(e) => e.stopPropagation()}>
                  <button
                    type="button"
                    className="iconbtn amenu-btn"
                    aria-haspopup="menu"
                    aria-expanded={cMenuOpen}
                    aria-label="More actions for this comment"
                    title="More actions"
                    onClick={() =>
                      setOpenMenuKey((cur) => (cur === cMenuKey ? null : cMenuKey))
                    }
                  >
                    <Icon name="dots" size="sm" />
                  </button>
                  {cMenuOpen && (
                    <div className="amenu" role="menu">
                      {cMine && !c.hidden && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          onClick={() => {
                            setOpenMenuKey(null)
                            setEditCmId(c.id)
                            setEditCmText(c.text)
                            setEditCmErr('')
                          }}
                        >
                          <Icon name="edit" size="sm" />
                          <span>Edit comment</span>
                        </button>
                      )}
                      {!cMine && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          disabled={cReported}
                          onClick={() => {
                            setOpenMenuKey(null)
                            openModal({
                              kind: 'forum-report',
                              postId: f.id,
                              commentId: c.id,
                            })
                          }}
                        >
                          <Icon name="flag" size="sm" />
                          <span>{cReported ? 'Reported' : 'Report comment'}</span>
                        </button>
                      )}
                      {isStaffUser && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item"
                          onClick={() => {
                            setOpenMenuKey(null)
                            toggleHideComment(f.id, c.id)
                          }}
                        >
                          <Icon name="eye" size="sm" />
                          <span>{c.hidden ? 'Restore comment' : 'Hide comment'}</span>
                        </button>
                      )}
                      {(cMine || isStaffUser) && (
                        <button
                          type="button"
                          role="menuitem"
                          className="amenu-item danger"
                          onClick={() => {
                            setOpenMenuKey(null)
                            deleteForumComment(f.id, c.id)
                          }}
                        >
                          <Icon name="trash" size="sm" />
                          <span>Delete comment</span>
                        </button>
                      )}
                    </div>
                  )}
                </div>
              )}
            </div>

            {!c.deleted && (
              <div className="tiny muted fcm-acts" title={ago(c.ts)}>
                {stamp(c.ts)}
                {c.edited && (
                  <>
                    {' '}
                    ·{' '}
                    <span className="edited-tag" title={`Edited ${stamp(c.edited)}`}>
                      edited
                    </span>
                  </>
                )}
                {!c.hidden && depth < 5 && (
                  <>
                    {' '}
                    ·{' '}
                    <button
                      type="button"
                      className="link"
                      onClick={() => {
                        setReplyToId((cur) => (cur === c.id ? null : c.id))
                        setReplyText('')
                      }}
                    >
                      Reply
                    </button>
                  </>
                )}
              </div>
            )}

            {replyToId === c.id && (
              <form
                className="row"
                style={{ gap: 8, marginTop: 6 }}
                onSubmit={(e) => {
                  e.preventDefault()
                  if (addForumComment(f.id, c.id, replyText)) {
                    setReplyText('')
                    setReplyToId(null)
                  }
                }}
              >
                <input
                  className="input"
                  value={replyText}
                  placeholder={`Reply to ${c.by}...`}
                  maxLength={500}
                  aria-label="Write a reply"
                  autoComplete="off"
                  onChange={(e) => setReplyText(e.target.value)}
                />
                <button type="submit" className="btn sm" aria-label="Post reply">
                  <Icon name="send" size="sm" />
                </button>
                <button
                  type="button"
                  className="btn ghost sm"
                  onClick={() => {
                    setReplyToId(null)
                    setReplyText('')
                  }}
                >
                  Cancel
                </button>
              </form>
            )}

            {kids && (
              <div className={`cthread ${depth >= 3 ? 'flat' : ''}`.trim()}>
                {kids}
              </div>
            )}
          </div>
        </div>
      )
    })
  }

  return (
    <div className="fcol">
      <button
        type="button"
        className="btn ghost sm"
        style={{ alignSelf: 'flex-start' }}
        onClick={() => navigate('forum')}
      >
        <Icon name="back" size="sm" />
        Back to the forum
      </button>

      <article
        className={`card stack fpost ${f.pinned ? 'pinned' : ''} ${
          f.hidden ? 'is-hidden' : ''
        }`.trim()}
        id={`post-${f.id}`}
        style={{ gap: 12 }}
      >
        <div className="fhead">
          <AvatarBadge
            account={f.kind !== 'notice' ? author : undefined}
            name={by}
            isStaff={isStaffPost}
            size="md"
          />
          <div className="grow">
            <div className="row wrap" style={{ gap: 6 }}>
              <b style={{ fontSize: 14 }}>{by}</b>
              {isStaffPost && <span className="pill navy">Staff</span>}
              {f.kind === 'notice' && <span className="pill yellow">Notice</span>}
              {(f.kind === 'student' || f.kind === 'school') && (
                <span className={`srcpill ${f.kind}`}>
                  {f.kind === 'school' ? 'Official RTU Store' : 'Student seller'}
                </span>
              )}
              {f.topic && f.kind === 'post' && <span className="pill">{f.topic}</span>}
              {f.pinned && <span className="pill yellow">Pinned</span>}
              {f.hidden && <span className="pill red">Hidden by staff</span>}
              {nrep > 0 && (
                <button
                  type="button"
                  className="pill red linkpill"
                  title="Open Forum Reports"
                  onClick={() => navigate('forum-reports')}
                >
                  {nrep} open report{nrep === 1 ? '' : 's'}
                </button>
              )}
            </div>
            <div className="tiny muted" title={ago(f.ts)}>
              {stamp(f.ts)} · {ago(f.ts)}
              {f.edited && (
                <>
                  {' '}
                  ·{' '}
                  <span className="edited-tag" title={`Edited ${stamp(f.edited)}`}>
                    edited
                  </span>
                </>
              )}
            </div>
          </div>

          <div className="amenu-wrap" onClick={(e) => e.stopPropagation()}>
            <button
              type="button"
              className="iconbtn amenu-btn"
              aria-haspopup="menu"
              aria-expanded={postMenuOpen}
              aria-label="More actions for this post"
              title="More actions"
              onClick={() =>
                setOpenMenuKey((cur) => (cur === postMenuKey ? null : postMenuKey))
              }
            >
              <Icon name="dots" size="sm" />
            </button>
            {postMenuOpen && (
              <div className="amenu" role="menu">
                {mine && f.kind === 'post' && !f.hidden && (
                  <button
                    type="button"
                    role="menuitem"
                    className="amenu-item"
                    onClick={() => {
                      setOpenMenuKey(null)
                      openModal({ kind: 'forum-edit-post', postId: f.id })
                    }}
                  >
                    <Icon name="edit" size="sm" />
                    <span>Edit post</span>
                  </button>
                )}
                {!mine && f.kind !== 'notice' && (
                  <button
                    type="button"
                    role="menuitem"
                    className="amenu-item"
                    disabled={reported}
                    onClick={() => {
                      setOpenMenuKey(null)
                      openModal({
                        kind: 'forum-report',
                        postId: f.id,
                        commentId: null,
                      })
                    }}
                  >
                    <Icon name="flag" size="sm" />
                    <span>{reported ? 'Reported' : 'Report post'}</span>
                  </button>
                )}
                {isStaffUser && (
                  <button
                    type="button"
                    role="menuitem"
                    className="amenu-item"
                    onClick={() => {
                      setOpenMenuKey(null)
                      togglePinPost(f.id)
                    }}
                  >
                    <Icon name="pin" size="sm" />
                    <span>{f.pinned ? 'Unpin post' : 'Pin post'}</span>
                  </button>
                )}
                {isStaffUser && (
                  <button
                    type="button"
                    role="menuitem"
                    className="amenu-item"
                    onClick={() => {
                      setOpenMenuKey(null)
                      toggleHidePost(f.id)
                    }}
                  >
                    <Icon name="eye" size="sm" />
                    <span>{f.hidden ? 'Restore post' : 'Hide post'}</span>
                  </button>
                )}
                {(mine || isStaffUser) && (
                  <button
                    type="button"
                    role="menuitem"
                    className="amenu-item danger"
                    onClick={() => {
                      setOpenMenuKey(null)
                      deleteForumPost(f.id)
                    }}
                  >
                    <Icon name="trash" size="sm" />
                    <span>Delete post</span>
                  </button>
                )}
              </div>
            )}
          </div>
        </div>

        {f.hidden && (
          <div
            className="note"
            role="status"
            style={{ background: 'var(--red-bg)', borderColor: 'transparent' }}
          >
            <Icon name="info" />
            <span>
              {isStaffUser
                ? 'This post is hidden from members.'
                : 'Staff hid this post. Only you and staff can see it.'}
            </span>
          </div>
        )}

        {f.kind === 'student' || f.kind === 'school' ? (
          <div className="fprod">
            {prodImgs.length > 0 && imgSrc(prodImgs[0]) && (
              <img
                className="fprod-img"
                src={imgSrc(prodImgs[0])}
                alt={f.title || ''}
              />
            )}
            <div className="grow" style={{ minWidth: 180 }}>
              <b>
                {f.svc ? 'New service: ' : 'New: '}
                {f.title}
              </b>
              <div className="small muted" style={{ overflowWrap: 'anywhere' }}>
                {bodyStr.length > 200 ? bodyStr.slice(0, 200) + '...' : bodyStr}
              </div>
              <div className="small tnum" style={{ marginTop: 2 }}>
                {liveProd ? priceLabel(liveProd) : peso(f.price || 0)}
                {f.tag ? ` · ${f.tag}` : ''}
              </div>
            </div>
            {liveProd ? (
              <button
                type="button"
                className="btn sm"
                onClick={() => openProduct(liveProd.id)}
              >
                View {f.svc ? 'service' : 'listing'}
              </button>
            ) : (
              <span className="pill">No longer available</span>
            )}
          </div>
        ) : (
          <>
            <h1 className="ptitle">{postTitle(f)}</h1>
            {!(
              f.kind === 'post' &&
              !String(f.title || '').trim() &&
              bodyStr.length <= 70
            ) && <div className="fbody full">{bodyStr}</div>}
          </>
        )}

        {postImgs.length > 0 && (
          <div className={`fimgs n${Math.min(postImgs.length, 4)} full`}>
            {postImgs.map((id, idx) => {
              const src = imgSrc(id)
              if (!src) {
                return (
                  <div key={`${id}-${idx}`} className="fimg-miss">
                    <Icon name="info" size="sm" />
                    <span>Image unavailable</span>
                  </div>
                )
              }
              return (
                <button
                  key={`${id}-${idx}`}
                  type="button"
                  className="fimg"
                  aria-label={`Open photo ${idx + 1} of ${postImgs.length}`}
                  onClick={() =>
                    openModal({ kind: 'forum-photo', postId: f.id, index: idx })
                  }
                >
                  <img
                    src={src}
                    alt={`Photo ${idx + 1} attached to the post ${postTitle(f)}`}
                    loading="lazy"
                  />
                </button>
              )
            })}
          </div>
        )}

        <div className="fcount">
          <span>
            {nReact ? `${nReact} reaction${nReact === 1 ? '' : 's'}` : 'No reactions yet'}
          </span>
          <span>
            {nCm ? `${nCm} comment${nCm === 1 ? '' : 's'}` : 'No comments yet'}
          </span>
        </div>

        <div className="freacts" role="group" aria-label="React to this post">
          {REACTS.map((r) => {
            const myReact = f.reacts[currentAccount.email]
            const n = Object.values(f.reacts || {}).filter((x) => x === r[0]).length
            return (
              <button
                key={r[0]}
                type="button"
                className={`rbtn ${myReact === r[0] ? 'on' : ''}`.trim()}
                aria-pressed={myReact === r[0]}
                aria-label={`${r[2]}${n ? `, ${n}` : ''}`}
                onClick={() => toggleForumReaction(f.id, r[0])}
              >
                <span className="e">{r[1]}</span>
                {n ? <span className="tnum">{n}</span> : <span className="muted">{r[2]}</span>}
              </button>
            )
          })}
          <span className="grow" />
        </div>

        {isStaffUser && (
          <div className="row wrap" style={{ gap: 6 }}>
            <span className="tiny muted">
              Staff tools: pin, hide and delete are in the three-dot menu.
            </span>
            <button
              type="button"
              className="btn ghost sm"
              onClick={() => navigate('forum-reports')}
            >
              Forum Reports
            </button>
          </div>
        )}
      </article>

      <section className="card stack" aria-label="Comments" style={{ gap: 12 }}>
        <b>
          Comments <span className="tiny muted tnum">({nCm})</span>
        </b>
        <form
          className="row"
          style={{ gap: 8 }}
          onSubmit={(e) => {
            e.preventDefault()
            if (addForumComment(f.id, null, newComment)) {
              setNewComment('')
            }
          }}
        >
          <AvatarBadge account={currentAccount} size="sm" />
          <input
            className="input"
            id={`cm-${f.id}`}
            value={newComment}
            placeholder="Add a comment..."
            maxLength={500}
            aria-label="Add a comment"
            autoComplete="off"
            onChange={(e) => setNewComment(e.target.value)}
          />
          <button type="submit" className="btn sm" aria-label="Post comment">
            <Icon name="send" size="sm" />
          </button>
        </form>

        {(f.comments || []).some((c) => !c.parent) ? (
          renderCommentTree(null, 0)
        ) : (
          <p className="small muted">No comments yet. Start the discussion.</p>
        )}
      </section>
    </div>
  )
}
