/**
 * TUNING CONFIG â€” Adjust all BrandLoader timings from this single object.
 * All durations are in seconds; thresholds are in milliseconds.
 */
export const LOADER_TIMING = {
  // Anti-flicker & readiness thresholds (ms)
  antiFlickerDelayMs: 150, // Never show if ready faster than this
  slowLoadThresholdMs: 8000, // Show "Taking longer than usual..." + Retry

  // Full intro build (~2.0s total)
  full: {
    tindahanDuration: 0.55,
    taglineLinesDuration: 0.45,
    tileDropDuration: 0.32,
    tileStagger: 0.035, // ~40ms stagger across 17 tiles
    tilesOverlap: "-=0.25", // Slight overlap with step 1
    letterDropDuration: 0.35,
    letterStagger: 0.11,
    cursorPopDuration: 0.2,
    cursorClickDuration: 0.15,
    tagSwingDuration: 0.45,
    exitDuration: 0.45,
  },

  // Compact variant (~0.9s build, ~0.3s exit) for repeat loads in same session
  compact: {
    timeScale: 2.2,
    exitDuration: 0.3,
  },

  // Fast-forward multiplier when app becomes ready mid-build (legacy fallback)
  fastForwardTimeScale: 3.5,

  // Adaptive exit durations based on actual load speed (seconds)
  adaptiveExitFast: 0.24,   // < 400ms load
  adaptiveExitNormal: 0.32, // 400â€“1200ms load
  adaptiveExitSlow: 0.42,   // > 1200ms load

  // Fiesta banderitas: wrapper divs drop in staggered, never flags/SVG
  bunting: {
    dropDuration: 0.7,
    topStart: 0.1,
    lowStart: 0.3,
    bottomStart: 0.45,
    dropDistance: 90, // px
  },

  // Calm idle shine sweep interval (seconds)
  idleShineDuration: 0.75,
  idleShineRepeatDelay: 1.6,

  // Reduced-motion fade duration (seconds)
  reducedMotionFade: 0.2,
} as const;

const SESSION_KEY = "rtu_tindahan_intro_seen";

// 17 awning tiles sampled left-to-right from the RTU Tindahan logo
const AWNING_COLORS = [
  "#ffbd59", "#c25959", "#ffffff",
  "#ffbd59", "#c25959", "#ffffff",
  "#ffbd59", "#c25959", "#ffffff",
  "#ffbd59", "#c25959", "#ffffff",

export type BrandLoaderVariant = "full" | "compact" | "auto";

export interface BrandLoaderProps {
  /** True when the app/page data is still loading */
  isLoading: boolean;
  /** "auto" plays full on first session visit, compact on subsequent boots */
  variant?: BrandLoaderVariant;
  /** Optional retry callback for the 8s slow-load fallback */
  onRetry?: () => void;
  /** Called after the exit animation finishes and the loader unmounts */
  onExited?: () => void;
  /** GSAP timeScale multiplier for build & exit (1 = normal, 2 = 2x faster, etc.) */
  timeScale?: number;
  /** Bypass the 150ms anti-flicker delay and show immediately (e.g. login flow) */
  immediate?: boolean;
}

function markIntroSeen(): void {
  try {
    window.sessionStorage.setItem(SESSION_KEY, "1");
  } catch {
    // Ignore storage errors in restricted/private contexts
  }
}

export const BrandLoader: React.FC<BrandLoaderProps> = ({
  isLoading,
  variant = "auto",
  onRetry,
  onExited,
  timeScale = 1,
  immediate = false,
}) => {
  const [shouldRender, setShouldRender] = useState<boolean>(false);
  const [isExiting, setIsExiting] = useState<boolean>(false);
  const [isSlowLoad, setIsSlowLoad] = useState<boolean>(false);

  const overlayRef = useRef<HTMLDivElement | null>(null);
  const stageRef = useRef<HTMLDivElement | null>(null);
  const buildTlRef = useRef<gsap.core.Timeline | null>(null);
  const idleTlRef = useRef<gsap.core.Timeline | null>(null);
  const buildCompletedRef = useRef<boolean>(false);
  const exitStartedRef = useRef<boolean>(false);
  const isLoadingRef = useRef<boolean>(isLoading);
  /** Recorded the moment shouldRender becomes true â€” used to compute elapsed load time. */
  const loadStartMsRef = useRef<number>(0);
  /** True on the very first render. On initial mount with isLoading=true we skip
   * the anti-flicker delay so the brand loader always shows on a fresh page load. */
  const initialMountRef = useRef(true);
  /** Prop refs â€” updated via effects so the build effect (which depends only on
   * shouldRender) always reads the latest props without re-running. */
  const variantRef = useRef<BrandLoaderVariant>(variant);
  const onExitedRef = useRef<(() => void) | undefined>(onExited);
  const timeScaleRef = useRef<number>(timeScale);
  /** Adaptive exit duration in seconds, set by accelerateToFinish(). */
  const adaptiveExitSecRef = useRef<number>(0.32);

  useEffect(() => {
    isLoadingRef.current = isLoading;
  }, [isLoading]);

  // Keep prop refs in sync so the build effect (which depends only on shouldRender)
  // always reads the latest props without re-running.
  useEffect(() => {
    variantRef.current = variant;
  }, [variant]);
  useEffect(() => {
    onExitedRef.current = onExited;
  }, [onExited]);
  useEffect(() => {
    timeScaleRef.current = timeScale;
  }, [timeScale]);

  // On initial mount with isLoading=true we show immediately â€” a full page reload
  // should always play the brand intro. When `immediate` is true (login flow) we
  // also bypass the delay so fast logins don't delay or skip the animation.
  // Subsequent loading states (route changes, data refetches) still get the
  // 150ms debounce to avoid flicker.
  useEffect(() => {
    if (!isLoading) return;
    if (shouldRender) return;

    if (initialMountRef.current || immediate) {
      loadStartMsRef.current = Date.now();
      setShouldRender(true);
      initialMountRef.current = false;

      if (isLoadingRef.current) {
        setShouldRender(true);
      }

      return;

      }

      },

      tl.to(overlayRef.current, { opacity: 0, duration: exitDuration, ease: "power1.out" });
    } else {

        y: -24,
        scale: 1.05,
        opacity: 0,
        duration: exitDuration * 0.8,
        ease: "power2.in",
      }).to(
        overlayRef.current,
        {
          opacity: 0,
          duration: exitDuration * 0.75,
          ease: "power2.out",
        },
        `-=${exitDuration * 0.45}`
      );
    }
  };

  // 5. GSAP Build & Idle Timeline (StrictMode-safe via gsap.context)
  //    Depends ONLY on [shouldRender] so prop changes (variant/timeScale/onExited)

  useLayoutEffect(() => {

    // Resolve variant once for this run â€” never re-resolve on prop changes.
    const currentVariant =
      variantRef.current !== "auto"
        ? variantRef.current
        : (() => {
            try {
              return window.sessionStorage.getItem(SESSION_KEY)
                ? "compact"
                : "full";
            } catch {
              return "full";
            }
          })();

      "(prefers-reduced-motion: reduce)"
    ).matches;

          ".bl-tagline-line-right",
          ".bl-tile",
          ".bl-letter",
          ".bl-cursor",
          ".bl-tag-group",

      }

      gsap.set(".bl-tagline-line-left", { scaleX: 0, transformOrigin: "right center" });
      gsap.set(".bl-tagline-line-right", { scaleX: 0, transformOrigin: "left center" });
      gsap.set(".bl-tile", { y: -32, scaleY: 0.75, opacity: 0, transformOrigin: "center bottom" });

const STEP_SECONDS = 3 // how long each image rests in the center
const SLIDE_SECONDS = 1.2 // how long the glide to the next image takes

/**
 * Auto-sliding, seamlessly looping carousel (no controls).
 *
 * Based on the GSAP "infinite cards" loop: every card flies across the track
 * while scaling/fading in and out (yoyo), staggered so the card in the middle
 * is biggest and its neighbours shrink away on each side. Instead of a
 * ScrollTrigger, a single `pos` value is eased from slide to slide, and every
 * card's pass is rendered from that value, so the loop never has a seam.
 */
function LoginCarousel({ images }: { images: { src: string; alt: string }[] }) {
  const rootRef = useRef<HTMLDivElement>(null)

  useLayoutEffect(() => {
    const root = rootRef.current
    if (!root) return

    const ctx = gsap.context(() => {
      const cards = gsap.utils.toArray<HTMLElement>('.login-carousel-card')
      const n = cards.length
      if (n === 0) return

      // One paused "pass" timeline per card: scale/opacity/grayscale peak in the
      // middle (yoyo) while the card travels right â†’ left in a straight line.
      const passes = cards.map((card) =>
        gsap
          .timeline({ paused: true })
          .fromTo(
            card,
            { scale: 0 },
            { scale: 1, duration: 0.5, yoyo: true, repeat: 1, ease: 'power1.in' },
            0
          )
          .fromTo(
            card,
            { opacity: 0, filter: 'grayscale(60%)' },
            { opacity: 1, filter: 'grayscale(0%)', duration: 0.5, yoyo: true, repeat: 1, ease: 'sine.in' },
            0
          )
          .fromTo(card, { xPercent: TRAVEL }, { xPercent: -TRAVEL, duration: 1, ease: 'none' }, 0)
      )

      const wrap = gsap.utils.wrap(0, 1)
      const state = { pos: 0 } // card `i` is centered whenever pos === i (mod n)

      const render = () => {
        passes.forEach((pass, i) => {
          const progress = wrap(0.5 + (state.pos - i) / n)
          pass.progress(progress)
          // keep the centered card on top (z-index must be an integer)
          cards[i].style.zIndex = String(Math.round((1 - Math.abs(progress - 0.5) * 2) * 100))
        })
      }
      render()

      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return

      let target = 0
      const advance = () => {
        target += 1
        gsap.to(state, {
          pos: target,
          duration: SLIDE_SECONDS,
          ease: 'power3.inOut',
          overwrite: true,
          onUpdate: render,
        })
      }

      // slide on a steady beat, forever
      gsap
        .timeline({ repeat: -1, delay: STEP_SECONDS })
        .call(advance)
        .to({}, { duration: STEP_SECONDS })
    }, root)

    return () => ctx.revert()
  }, [images.length])

  return (
    <div className="login-carousel" ref={rootRef} aria-roledescription="carousel" aria-label="Campus marketplace images">
      <div className="login-carousel-track">
        {images.map(({ src, alt }) => (
          <div className="login-carousel-card" key={src}>
            <img src={src} alt={alt} draggable={false} />
          </div>
        ))}
      </div>
    </div>
  )
}

/**
 * AUTH CONTEXT â€” the Django backend is the source of truth
 * ------------------------------------------------------------
 * Replaced the prototype's offline authentication. Now:
 *
 *   Sign up   1. details  -> POST /auth/register/  (creates a visitor account
 *                           and emails a 6-digit code)
 *             2. code     -> POST /auth/verify-code/ (activates the account and
 *                           returns a token, so the user is signed in)
 *   Sign in                -> POST /auth/login/  (token, or an error code such
 *                           as "email_not_verified")
 *   Reload                 -> GET /auth/me/ with the stored token
 *   Sign out               -> POST /auth/logout/ deletes the token server-side
 *
 * The token lives in localStorage under `rtu_token` (see common/api/client.ts)
 * and is sent as `Authorization: Token <key>`.
 */

const STUDENT_EMAIL = /^\d{4}-\d{6}@rtu\.edu\.ph$/i
const LOGIN_EMAIL = /^[^@\s]+@rtu\.edu\.ph$/i

/** What the sign-up screen collects before the code is sent. */
export interface RegisterInput {
  first: string
  last: string
  email: string
  pass: string
  prog: string
  year: number | null
  consent: boolean
}

interface OtpState {
  /** Where the code was sent. Drives the "we emailed you" screen. */
  to: string
  /** When it was requested, so the screen can show a resend countdown. */
  sentAt: number
}

interface AuthValue {
  user: User | null
  /** true while restoring the session on first load */
  isLoading: boolean
  /** true while the login() API call is in flight (drives the local submit button spinner) */
  isSigningIn: boolean
  /** true after a successful login, drives the BrandLoader transition into the marketplace */
  isTransitioning: boolean
  /** Clears isTransitioning so BrandLoader can exit after the transition */
  clearTransitioning: () => void
  /** The pending sign-up code request (null before one has been sent) */
  otp: OtpState | null
  /** Clear the pending code request (used when starting over on step 1). */
  clearOtp: () => void
  login: (email: string, password: string) => Promise<void>
  /** Step 2 of sign up: create the account and email a 6-digit code. */
  register: (input: RegisterInput) => Promise<void>
  /** Ask for another 6-digit code (30-second cooldown server-side). */
  resendCode: (email: string) => Promise<void>
  /** Step 3 of sign up: check the code and sign in. */
  confirmCode: (email: string, code: string) => Promise<void>
  logout: () => Promise<void>
}

const AuthContext = createContext<AuthValue | null>(null)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null)
  const [isLoading, setIsLoading] = useState<boolean>(true)
  const [isSigningIn, setIsSigningIn] = useState<boolean>(false)
  const [isTransitioning, setIsTransitioning] = useState<boolean>(false)
  const [otp, setOtp] = useState<OtpState | null>(null)

  // -----------------------------------------------------------------------
  // On first load: if a token is stored, ask the server who it belongs to.
  // A rejected token is dropped so the user simply appears signed out.
  // -----------------------------------------------------------------------
  useEffect(() => {
    let cancelled = false
    const restore = async () => {
      if (!getToken()) {
        setIsLoading(false)
        return
      }
      try {
        const me = await apiFetch<User>('/auth/me/')
        if (!cancelled) setUser(me)
      } catch {
        setToken(null)
      } finally {
        if (!cancelled) setIsLoading(false)
      }
    }
    void restore()
    return () => {
      cancelled = true
    }
  }, [])

  // -----------------------------------------------------------------------
  // Sign in
  // -----------------------------------------------------------------------
  const login = useCallback(async (email: string, password: string) => {
    const address = email.trim().toLowerCase()
    if (!LOGIN_EMAIL.test(address)) {
      throw new Error('Use your RTU email, like 2024-123456@rtu.edu.ph.')
    }
    if (!password) throw new Error('Enter your password.')

    setIsSigningIn(true)
    try {
      const data = await apiFetch<{ token: string; user: User }>('/auth/login/', {
        method: 'POST',
        body: { email: address, password },
      })
      setToken(data.token)
      setUser(data.user)
      setIsTransitioning(true)
    } finally {
      setIsSigningIn(false)
    }
  }, [])

  // -----------------------------------------------------------------------
  // Sign up, step 1 -> 2: create the account, then email the code
  // -----------------------------------------------------------------------
  const register = useCallback(async (input: RegisterInput) => {
    const address = input.email.trim().toLowerCase()
    if (!STUDENT_EMAIL.test(address)) {
      throw new Error('Your RTU email must look like 2024-123456@rtu.edu.ph.')
    }
    if (!input.consent) {
      throw new Error('Accept the privacy notice to create your account.')
    }

    const displayName = `${input.first.trim()} ${input.last.trim()}`.trim()
    await apiFetch<{ detail: string; email: string }>('/auth/register/', {
      method: 'POST',
      body: {
        email: address,
        display_name: displayName,
        password: input.pass,
        password_confirm: input.pass,
        member_type: 'student',
        college: input.prog,
        ...(input.year ? { year_level: input.year } : {}),
        privacy_consent: input.consent,
      },
    })

    setOtp({ to: address, sentAt: Date.now() })
  }, [])

  const resendCode = useCallback(async (email: string) => {
    const address = email.trim().toLowerCase()
    await apiFetch<{ detail: string }>('/auth/resend-code/', {
      method: 'POST',
      body: { email: address },
    })
    setOtp({ to: address, sentAt: Date.now() })
  }, [])

  // -----------------------------------------------------------------------
  // Sign up, step 3: the code activates the account and signs the user in
  // -----------------------------------------------------------------------
  const confirmCode = useCallback(async (email: string, code: string) => {
    const address = email.trim().toLowerCase()
    const digits = String(code).replace(/\s/g, '')
    if (!/^\d{6}$/.test(digits)) {
      throw new Error('Enter the 6 digits from the email.')
    }

    const data = await apiFetch<{ token: string; user: User }>(
      '/auth/verify-code/',
      { method: 'POST', body: { email: address, code: digits } },
    )
    setToken(data.token)
    setUser(data.user)
    setOtp(null)
  }, [])

  // Clear the pending code request so starting over on step 1 does not
  // auto-resume on the code screen.
  const clearOtp = useCallback(() => {
    setOtp(null)
  }, [])

  const clearTransitioning = useCallback(() => {
    setIsTransitioning(false)
  }, [])

  // -----------------------------------------------------------------------
  // Sign out
  // -----------------------------------------------------------------------
  const logout = useCallback(async () => {
    try {
      await apiFetch('/auth/logout/', { method: 'POST' })
    } catch {
      /* the token may already be gone; local state still has to clear */
    }
    setToken(null)
    setUser(null)
    setOtp(null)
  }, [])

  const value = useMemo(
    () => ({
      user,
      isLoading,
      isSigningIn,
      isTransitioning,
      otp,
      clearOtp,
      clearTransitioning,
      login,
      register,
      resendCode,
      confirmCode,
      logout,
    }),
    [user, isLoading, isSigningIn, isTransitioning, otp, clearOtp, clearTransitioning, login, register, resendCode, confirmCode, logout],
  )
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

export function useAuth(): AuthValue {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error('useAuth must be used inside <AuthProvider>')
  return ctx
}
